// Temporary fixture-only timings. No query text, argument, identity, evidence,
// credential, raw error or wall-clock timestamp is retained. Timings include
// scheduling and all work inside a port; they do not establish SQL lock time.
// Each fixed stage has its own last sixteen completions and first error so the
// forty unchanged warm polls cannot hide ingress. Calls keep their exact
// original context, inputs, results and marker facets. No background work.
type nativeDiagnosticStage uint8

const (
	nativeDiagnosticOff nativeDiagnosticStage = iota
	nativeDiagnosticPreparation
	nativeDiagnosticMetadata
	nativeDiagnosticHTTP
	nativeDiagnosticPublication
)

func (s nativeDiagnosticStage) name() string {
	switch s {
	case nativeDiagnosticPreparation:
		return "preparation"
	case nativeDiagnosticMetadata:
		return "metadata"
	case nativeDiagnosticHTTP:
		return "http"
	case nativeDiagnosticPublication:
		return "publication"
	default:
		return "off"
	}
}

type nativeDiagnosticEvent struct {
	phase                                                                         string
	start, elapsed, remainingStart, remainingEnd, inputRemaining, resultRemaining time.Duration
	result, contextResult                                                         string
}
type nativeDiagnosticTotal struct {
	calls   uint64
	elapsed time.Duration
}
type nativeDiagnosticBuffer struct {
	events     [16]nativeDiagnosticEvent
	next, used int
	firstError *nativeDiagnosticEvent
	totals     map[string]nativeDiagnosticTotal
}
type nativePhaseDiagnostic struct {
	mu     sync.Mutex
	start  time.Time
	active nativeDiagnosticStage
	stages [5]nativeDiagnosticBuffer
}

func newNativePhaseDiagnostic(t *testing.T) *nativePhaseDiagnostic {
	s := &nativePhaseDiagnostic{start: time.Now()}
	t.Cleanup(func() { s.log(t) })
	return s
}
func (s *nativePhaseDiagnostic) stage(v nativeDiagnosticStage) {
	if s == nil {
		return
	}
	if v > nativeDiagnosticPublication {
		v = nativeDiagnosticOff
	}
	s.mu.Lock()
	s.active = v
	s.mu.Unlock()
}
func nativeDiagnosticResult(err error) string {
	switch {
	case err == nil:
		return "ok"
	case errors.Is(err, context.DeadlineExceeded):
		return "context-deadline"
	case errors.Is(err, context.Canceled):
		return "context-canceled"
	case errors.Is(err, artifact.ErrDownloadExpired):
		return "download-expired"
	case errors.Is(err, artifact.ErrDownloadUnavailable):
		return "download-unavailable"
	case errors.Is(err, artifact.ErrDownloadConflict):
		return "download-conflict"
	case errors.Is(err, artifact.ErrDownloadCapacity):
		return "download-capacity"
	case errors.Is(err, artifact.ErrInvalid):
		return "invalid"
	case errors.Is(err, artifact.ErrIntegrity):
		return "integrity"
	case errors.Is(err, identity.ErrDenied):
		return "identity-denied"
	case errors.Is(err, identity.ErrUnavailable):
		return "identity-unavailable"
	default:
		return "other-error"
	}
}
func nativeDiagnosticRemaining(until, now time.Time) time.Duration {
	if until.IsZero() {
		return 0
	}
	return until.Sub(now)
}
func (s *nativePhaseDiagnostic) begin(ctx context.Context, phase string, inputUntil time.Time) func(error, time.Time) {
	if s == nil {
		return func(error, time.Time) {}
	}
	s.mu.Lock()
	stage := s.active
	s.mu.Unlock()
	if stage == nativeDiagnosticOff {
		return func(error, time.Time) {}
	}
	phase = nativeDiagnosticLabel(phase)
	start := time.Now()
	var deadline time.Time
	if ctx != nil {
		deadline, _ = ctx.Deadline()
	}
	return func(err error, until time.Time) {
		end := time.Now()
		e := nativeDiagnosticEvent{phase: phase, start: start.Sub(s.start), elapsed: end.Sub(start), remainingStart: nativeDiagnosticRemaining(deadline, start), remainingEnd: nativeDiagnosticRemaining(deadline, end), inputRemaining: nativeDiagnosticRemaining(inputUntil, end), resultRemaining: nativeDiagnosticRemaining(until, end), result: nativeDiagnosticResult(err), contextResult: "none"}
		if ctx != nil {
			e.contextResult = nativeDiagnosticResult(ctx.Err())
		}
		s.mu.Lock()
		defer s.mu.Unlock()
		b := &s.stages[stage]
		b.events[b.next] = e
		b.next = (b.next + 1) % len(b.events)
		b.used = min(b.used+1, len(b.events))
		if err != nil && b.firstError == nil {
			first := e
			b.firstError = &first
		}
		if b.totals == nil {
			b.totals = make(map[string]nativeDiagnosticTotal)
		}
		total := b.totals[phase]
		total.calls++
		total.elapsed += e.elapsed
		b.totals[phase] = total
	}
}
func (s *nativePhaseDiagnostic) log(t *testing.T) {
	if s == nil {
		return
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	for stage := nativeDiagnosticPreparation; stage <= nativeDiagnosticPublication; stage++ {
		b := &s.stages[stage]
		logEvent := func(kind string, e nativeDiagnosticEvent) {
			t.Logf("native bounded phase=%s kind=%s port=%s start=%s elapsed=%s remaining-start=%s remaining-end=%s input-remaining=%s result-remaining=%s result=%s context=%s", stage.name(), kind, e.phase, e.start, e.elapsed, e.remainingStart, e.remainingEnd, e.inputRemaining, e.resultRemaining, e.result, e.contextResult)
		}
		for i := 0; i < b.used; i++ {
			logEvent("recent", b.events[(b.next-b.used+i+len(b.events))%len(b.events)])
		}
		if b.firstError != nil {
			logEvent("first-error", *b.firstError)
		}
		for _, label := range nativeDiagnosticLabels {
			if v, ok := b.totals[label]; ok {
				t.Logf("native bounded phase=%s kind=total port=%s calls=%d elapsed=%s", stage.name(), label, v.calls, v.elapsed)
			}
		}
	}
}

func (s *nativePhaseDiagnostic) authority(a authapp.ContextDownloadAuthority) evidence.CurrentDownloadAuthority {
	a.Permissions = nativePhasePermissions{a.Permissions, s}
	a.Catalog = nativePhaseCatalog{a.Catalog, s}
	a.Retention = nativePhaseRetention{a.Retention, s}
	a.Environment = nativePhaseEnvironment{a.Environment, s}
	return nativePhaseAuthority{a, s}
}
func (s *nativePhaseDiagnostic) runtime(t *testing.T, c evidence.ContextNativeRuntimeConfig) evidence.ContextNativeRuntimeConfig {
	t.Helper()
	a, ok := c.Authority.(authapp.ContextDownloadAuthority)
	if !ok {
		t.Fatal("diagnostic requires original composed authority")
	}
	c.Authority = s.authority(a)
	c.Workers = nativePhaseAttachments{c.Workers, s}
	c.Preparation = nativePhasePreparation{c.Preparation, s}
	c.Audit = nativePhaseAudit{c.Audit, s}
	c.Acknowledgements = nativePhaseAcknowledgements{c.Acknowledgements, s}
	c.Native.Fences = nativePhaseFences{c.Native.Fences, s}
	owners, ok := c.Native.Owners.(nativePhaseFirstOperations)
	if !ok {
		t.Fatal("diagnostic requires original first-context operations")
	}
	c.Native.Owners = nativePhaseOperations{owners, s}
	c.Native.Origins.Operations = nativePhaseOperationReader{c.Native.Origins.Operations, s}
	c.Native.PreparationKeys = nativePhasePreparationKeys{c.Native.PreparationKeys, s}
	c.Native.Registrar = nativePhaseRegistrar{c.Native.Registrar, s}
	return c
}
func (s *nativePhaseDiagnostic) transfer(t *testing.T, in evidence.DownloadResponseTransfer) evidence.DownloadResponseTransfer {
	t.Helper()
	a, ok := in.Authority.(authapp.ContextDownloadAuthority)
	if !ok {
		t.Fatal("diagnostic requires original transfer authority")
	}
	r, ok := in.Responses.(nativePhaseHTTPFirstResponses)
	if !ok {
		t.Fatal("diagnostic requires original first-context responses")
	}
	in.Authority = s.authority(a)
	in.Sessions = nativePhaseHTTPSessions{in.Sessions, s}
	in.Responses = nativePhaseHTTPResponses{r, s}
	in.Objects = nativeDiagnosticWindowObjects{in.Objects, s}
	in.Audit = nativeDiagnosticWindowAck{in.Audit, s}
	return in
}

type nativePhaseRegistrar struct {
	evidence.NativeExchangeRegistrar
	trace *nativePhaseDiagnostic
}

func (s nativePhaseRegistrar) RegisterNativeExchange(ctx context.Context, in evidence.NativeExchangeRegistration, before func(context.Context, evidence.NativeOrigin) error, check func(context.Context, evidence.NativeOrigin) (value.NativeLease, error)) (evidence.NativePendingExchange, error) {
	done := s.trace.begin(ctx, "native.register", in.Operation.Until)
	v, e := s.NativeExchangeRegistrar.RegisterNativeExchange(ctx, in, func(call context.Context, origin evidence.NativeOrigin) error {
		finished := s.trace.begin(call, "native.before-command", origin.Lease.Until)
		err := before(call, origin)
		finished(err, time.Time{})
		return err
	}, func(call context.Context, origin evidence.NativeOrigin) (value.NativeLease, error) {
		finished := s.trace.begin(call, "native.current-frame", origin.Lease.Until)
		lease, err := check(call, origin)
		finished(err, lease.Until)
		return lease, err
	})
	done(e, time.Time{})
	if v != nil {
		v = nativePhasePending{v, s.trace}
	}
	return v, e
}

// Labels remain closed even if a failing implementation supplies an unexpected
// phase. That original value still reaches the underlying port unchanged.
func nativePhaseHTTPAuditPhase(phase, action string) string {
	switch phase {
	case "intent":
		switch action {
		case "claim":
			return "http.audit-intent-claim"
		case "ack":
			return "http.audit-intent-ack"
		case "record":
			return "http.audit-intent-record"
		case "read":
			return "http.audit-intent-read"
		}
	case "outcome":
		switch action {
		case "claim":
			return "http.audit-outcome-claim"
		case "ack":
			return "http.audit-outcome-ack"
		case "record":
			return "http.audit-outcome-record"
		case "read":
			return "http.audit-outcome-read"
		}
	}
	return "http.audit-invalid-phase"
}

type nativeDiagnosticAuthorityTest struct {
	ctx     context.Context
	current identities.CurrentIdentity
	scope   artifact.DownloadScope
	value   artifact.DownloadAuthority
	err     error
	calls   int
}

func (s *nativeDiagnosticAuthorityTest) CheckDownloadCurrent(ctx context.Context, c identities.CurrentIdentity, scope artifact.DownloadScope) (artifact.DownloadAuthority, error) {
	s.ctx, s.current, s.scope = ctx, c, scope
	s.calls++
	return s.value, s.err
}
func TestNativePhaseDiagnosticPreservesOriginalCallAndDeadline(t *testing.T) {
	ctx, cancel := context.WithDeadline(context.Background(), time.Now().Add(time.Second))
	defer cancel()
	unavailable := errors.New("private diagnostic sentinel never logged")
	in := &nativeDiagnosticAuthorityTest{value: artifact.DownloadAuthority{CheckedAt: time.Now(), ValidUntil: time.Now().Add(time.Millisecond)}, err: unavailable}
	s := &nativePhaseDiagnostic{start: time.Now()}
	s.stage(nativeDiagnosticPreparation)
	a := nativePhaseAuthority{in, s}
	scope := artifact.DownloadScope{ArtifactID: "original-not-logged"}
	v, err := a.CheckDownloadCurrent(ctx, nil, scope)
	if in.calls != 1 || in.ctx != ctx || in.current != nil || in.scope != scope || v != in.value || err != unavailable {
		t.Fatal("trace changed original authority call")
	}
	cancel()
	_, err = a.CheckDownloadCurrent(ctx, nil, scope)
	if err != unavailable || in.calls != 2 || in.ctx != ctx {
		t.Fatal("trace replaced canceled original call")
	}
	b := s.stages[nativeDiagnosticPreparation]
	if b.firstError == nil || b.firstError.result != "other-error" || b.firstError.remainingStart <= 0 || b.firstError.elapsed < 0 || b.events[1].contextResult != "context-canceled" {
		t.Fatal("incorrect bounded timing classification")
	}
}
func TestNativePhaseDiagnosticKeepsIndependentBoundedStages(t *testing.T) {
	s := &nativePhaseDiagnostic{start: time.Now()}
	s.stage(nativeDiagnosticPreparation)
	s.begin(context.Background(), "native.claim", time.Time{})(artifact.ErrDownloadConflict, time.Time{})
	var joined sync.WaitGroup
	for range 4 {
		joined.Add(1)
		go func() {
			defer joined.Done()
			for range 40 {
				s.begin(context.Background(), "authority.current", time.Time{})(nil, time.Time{})
			}
		}()
	}
	joined.Wait()
	s.stage(nativeDiagnosticOff)
	for range 40 {
		s.begin(context.Background(), "authority.current", time.Time{})(nil, time.Time{})
	}
	s.stage(nativeDiagnosticHTTP)
	s.begin(context.Background(), "http.audit-reserve", time.Time{})(artifact.ErrDownloadCapacity, time.Time{})
	prep, http := s.stages[nativeDiagnosticPreparation], s.stages[nativeDiagnosticHTTP]
	if prep.used != 16 || prep.firstError == nil || prep.firstError.phase != "native.claim" || prep.totals["authority.current"].calls != 160 || http.used != 1 || http.firstError.phase != "http.audit-reserve" {
		t.Fatal("trace lost independent first error or exceeded bound")
	}
	// Dynamic input can reach neither the totals key nor retained event/error.
	s.begin(context.Background(), "private-id-or-token", time.Time{})(errors.New("private body"), time.Time{})
	if s.stages[nativeDiagnosticHTTP].events[1].phase != "other" || s.stages[nativeDiagnosticHTTP].events[1].result != "other-error" || len(s.stages[nativeDiagnosticHTTP].totals) != 2 {
		t.Fatal("dynamic diagnostic label/error escaped classification")
	}
}
