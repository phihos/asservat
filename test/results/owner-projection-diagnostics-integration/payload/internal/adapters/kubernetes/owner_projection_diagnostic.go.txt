// Only closed codes are retained across the SQL normalizer. Neither original
// errors nor protected values enter this invocation-owned diagnostic slot.
type ownerProjectionDiagnosticSlot struct {
	first atomic.Pointer[ownerProjectionDiagnostic]
}

type ownerProjectionDiagnostic struct {
	check  ownerProjectionCheck
	cause  error
	inner  *ownerProjectionDiagnostic
	native *ownerOperationDiagnostic
}

func (e *ownerProjectionDiagnostic) Error() string {
	text := e.check.String()
	if e.inner != nil {
		text += ": " + e.inner.Error()
	}
	if e.native != nil {
		text += ": " + e.native.Error()
	}
	return text
}
func (e *ownerProjectionDiagnostic) Unwrap() error { return e.cause }
func projectionDiagnostic(check ownerProjectionCheck, cause error) *ownerProjectionDiagnostic {
	var inner *ownerProjectionDiagnostic
	var native *ownerOperationDiagnostic
	_ = errors.As(cause, &inner)
	_ = errors.As(cause, &native)
	return &ownerProjectionDiagnostic{check: check, cause: cause, inner: inner, native: native}
}
func projectionNativeCodes(e *ownerOperationDiagnostic) *ownerOperationDiagnostic {
	if e == nil {
		return nil
	}
	return &ownerOperationDiagnostic{check: e.check, detail: projectionNativeCodes(e.detail)}
}
func projectionFixedCodes(err error) *ownerProjectionDiagnostic {
	var e *ownerProjectionDiagnostic
	if !errors.As(err, &e) || e == nil {
		return nil
	}
	return &ownerProjectionDiagnostic{check: e.check, inner: projectionFixedCodes(e.inner), native: projectionNativeCodes(e.native)}
}
func (s *OwnerProjectionOperation) diagnosticReturn(check ownerProjectionCheck, err *error) {
	if *err == nil {
		return
	}
	if _, ok := (*err).(*ownerProjectionDiagnostic); !ok {
		*err = projectionDiagnostic(check, *err)
	}
	if s != nil && s.self == s && s.failure != nil {
		s.failure.first.CompareAndSwap(nil, projectionFixedCodes(*err))
	}
}

// The original inspection error remains the sole unwrap/disposition source.
// First-attempt codes explain the stopped invocation; they cannot widen its
// error classification, refill an attempt or request another inspection.
type ownerProjectionInspectionDiagnostic struct {
	cause             error
	first, inspection *ownerProjectionDiagnostic
}

func (e *ownerProjectionInspectionDiagnostic) Error() string {
	first, inspection := "projection.unknown", "projection.unknown"
	if e.first != nil {
		first = e.first.Error()
	}
	if e.inspection != nil {
		inspection = e.inspection.Error()
	}
	return "owner projection advance: " + first + "; inspection: " + inspection
}
func (e *ownerProjectionInspectionDiagnostic) Unwrap() error { return e.cause }
