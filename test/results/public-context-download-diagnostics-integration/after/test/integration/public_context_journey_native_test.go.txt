// The artifact backend is chosen BEFORE acquisition. The same protected wrapped
// keys are subsequently consumed through the actual remote three-slot A-KEY
// service; no captured object is re-encrypted or replaced for this fixture.
type publicContextJourneyKeys struct {
	scope   value.KeyScope
	seal    *kms.LocalKeyBackend
	backend *kms.LocalOperations
}

func newPublicContextJourneyKeys(t *testing.T, installation string) *publicContextJourneyKeys {
	scope, seal, backend := localDownloadKeys(t, installation)
	return &publicContextJourneyKeys{scope, seal, backend}
}
func publicJourneyRequire(t *testing.T, stage string, err error) {
	t.Helper()
	if err != nil {
		t.Fatal(stage, err)
	}
}

// This is precisely the existing synthetic Job/Pod fixture, serialized because
// the live requesting operation and the emulated acquisition process overlap.
// It does not invent a download/session: Create receives the actual SQL intent.
type publicJourneyJobs struct {
	mu    sync.Mutex
	inner syntheticDownloadJobs
}

func (j *publicJourneyJobs) snapshot() value.DownloadWorker {
	j.mu.Lock()
	defer j.mu.Unlock()
	return j.inner.worker
}
func (j *publicJourneyJobs) CreateDownloadWorker(c context.Context, v value.DownloadWorkerLaunch) (value.DownloadWorkerJob, error) {
	j.mu.Lock()
	defer j.mu.Unlock()
	return j.inner.CreateDownloadWorker(c, v)
}
func (j *publicJourneyJobs) FindDownloadWorkerJob(c context.Context, v value.DownloadWorkerIntent) (value.DownloadWorkerJob, error) {
	j.mu.Lock()
	defer j.mu.Unlock()
	return j.inner.FindDownloadWorkerJob(c, v)
}
func (j *publicJourneyJobs) ObserveDownloadWorker(c context.Context, v value.DownloadWorkerIntent, b value.DownloadWorkerJob) (value.DownloadWorker, error) {
	j.mu.Lock()
	defer j.mu.Unlock()
	return j.inner.ObserveDownloadWorker(c, v, b)
}
func (j *publicJourneyJobs) ObserveDownloadWorkerTermination(c context.Context, v value.DownloadWorker) (value.DownloadWorkerTermination, error) {
	j.mu.Lock()
	defer j.mu.Unlock()
	return j.inner.ObserveDownloadWorkerTermination(c, v)
}
func (j *publicJourneyJobs) RetireDownloadWorker(c context.Context, v value.DownloadWorkerTermination) (value.DownloadWorkerRetirement, error) {
	j.mu.Lock()
	defer j.mu.Unlock()
	return j.inner.RetireDownloadWorker(c, v)
}

type publicJourneyEnrollmentProof struct {
	jobs  *publicJourneyJobs
	token evidence.DownloadPeerToken
}

func (p publicJourneyEnrollmentProof) ReadWorkerEnrollmentProof(c context.Context, s, w string, k artifact.Digest) (value.WorkerEnrollmentProof, error) {
	worker := p.jobs.snapshot()
	if !worker.Valid() {
		return value.WorkerEnrollmentProof{}, artifact.ErrDownloadUnavailable
	}
	if p.token == nil {
		return (syntheticWorkerEnrollmentProof{worker}).ReadWorkerEnrollmentProof(c, s, w, k)
	}
	return (channelSQLProof{p.token, worker}).ReadWorkerEnrollmentProof(c, s, w, k)
}

type publicJourneyNativeProof struct {
	jobs  *publicJourneyJobs
	token evidence.DownloadPeerToken
}

func (p publicJourneyNativeProof) ReadDownloadPeerProof(c context.Context, s string, k value.KeyPeer) (value.KeyWorkerProof, error) {
	return (nativeSQLProof{p.jobs.snapshot(), p.token}).ReadDownloadPeerProof(c, s, k)
}

// All pools are purpose-specific existing fixture roles. Native readers, key
// ledgers, dispatch, enrollment and transfer are real product implementations;
// accepted bootstrap/backing, readiness, human/Pod proof, Job API and CA remain
// explicitly synthetic. This is NOT the protected installed OwnerRuntime graph.
type publicJourneyNative struct {
	services         bootstrap.ContextNativeDownloadServices
	coordinator      evidence.ContextDownloadCoordinator
	gate             authapp.ContextDownloadAuthority
	runtime, audit   *postgres.Downloads
	allocation       artifact.DownloadAllocation
	profile          value.DownloadWorkerProfile
	jobs             *publicJourneyJobs
	enrollment       workerEnrollmentFixture
	enrollmentRouter *evidence.WorkerEnrollmentRouter
	requestOwners    *postgres.WorkerOwners
	origins          evidence.NativeOrigins
	router           *native.HubRouter
	keyAuthority     *postgres.NativeKeyAuthority
	keys             *publicContextJourneyKeys
	store            *objectstore.Store
	storeID          string
	target           value.WorkerCredentialTarget
	joined           <-chan nativeSQLWorkerOutcome
	closeWorker      func()
	enrollmentTrace  *journeyEnrollmentTrace
}

// Set up the real credential channel after the first Prepare has actually
// dispatched its worker. Enroll waits for a real current-request owner; it never
// publishes one, preactivates a certificate or retries an uncertain POST.
func (f *publicJourneyNative) credentialClient(t *testing.T, ctx context.Context, session string) *mtls.WorkerCredentials {
	t.Helper()
	w := f.jobs.snapshot()
	if !w.Valid() || w.Intent.SessionID != session || w.Intent.Profile != f.profile {
		t.Fatal("actual first preparation did not dispatch its original worker")
	}
	f.target = value.WorkerCredentialTarget{InstallationID: w.Binding.InstallationID, SessionID: session, WorkerID: w.Intent.ID, PodUID: w.Binding.PodUID, CreatedAt: w.Intent.CreatedAt, RunUntil: w.Intent.RunUntil}
	h, e := workertransport.NewHandler(f.target.InstallationID, journeyEnrollmentPorts{WorkerEnrollmentExchange: f.enrollmentRouter, trace: f.enrollmentTrace}, time.Now, 1)
	publicJourneyRequire(t, "worker credential handler", e)
	server, roots := channelServer(t, h)
	ca := x509.NewCertPool()
	ca.AddCert(f.enrollment.issuer.ca)
	c, e := mtls.NewWorkerCredentials(server.URL, f.target, roots, ca, newWorkerChannelFixture(t), time.Now)
	publicJourneyRequire(t, "process credential client", e)
	t.Cleanup(c.Close)
	return c
}
func (f *publicJourneyNative) enrollAndStart(t *testing.T, ctx context.Context, c *mtls.WorkerCredentials) {
	t.Helper()
	e := wait.PollUntilContextTimeout(ctx, 10*time.Millisecond, time.Second, true, func(call context.Context) (bool, error) {
		_, err := f.requestOwners.ReadWorkerOwner(call, f.target.SessionID, f.target.WorkerID)
		return err == nil, nil
	})
	publicJourneyRequire(t, "actual requesting owner became current", e)
	err := c.Enroll(ctx)
	f.enrollmentTrace.Report(t)
	publicJourneyRequire(t, "actual routed enrollment", err)
	certificate := *channelCert(t, c)
	peer, e := c.Peer()
	publicJourneyRequire(t, "current worker peer", e)
	digest, e := c.PublicKeyDigest()
	publicJourneyRequire(t, "original worker public key", e)
	keyTLS := newKeyTLSFor(t, f.target.InstallationID, "artifact", f.keyAuthority, f.keys.backend, func(http.Handler) http.Handler {
		h, e := keyservice.NewHandler([]authapp.KeyService{{Scope: f.keys.scope, Authority: f.keyAuthority, Backend: fixtureConfiguredKeyBackend{f.keys.backend}, Now: time.Now}}, f.enrollment.issuer.verifier, time.Now, 2)
		publicJourneyRequire(t, "actual key handler", e)
		return h
	})
	keyTLS.server.TLS.ClientCAs.AddCert(f.enrollment.issuer.ca)
	transport := &http.Transport{TLSClientConfig: &tls.Config{RootCAs: keyTLS.roots, Certificates: []tls.Certificate{certificate}, MinVersion: tls.VersionTLS13}, DisableKeepAlives: true}
	t.Cleanup(transport.CloseIdleConnections)
	bound, e := f.store.ForCustody(f.storeID)
	publicJourneyRequire(t, "original artifact store", e)
	reader, e := objectstore.NewNativeContextReader(bound, f.keys.scope, keyTLS.server.URL, &http.Client{Transport: transport, Timeout: 5 * time.Second})
	publicJourneyRequire(t, "native reader", e)
	result, e := evidence.NewRegionalContextResult(ctx, f.target, reader, time.Now)
	publicJourneyRequire(t, "regional result", e)
	t.Cleanup(func() {
		result.Close()
		publicJourneyRequire(t, "regional result joined", result.Wait(context.Background()))
	})
	connectorCertificate := nativeSQLConnectorCertificate(t, nativeExchangeFixture{worker: f.jobs.snapshot(), enrollment: f.enrollment})
	connectorVerifier, e := mtls.NewPeerVerifier(keyTLS.roots, f.target.InstallationID, fleet.ConnectorPeer, time.Now)
	publicJourneyRequire(t, "connector TLS verifier", e)
	hub := &nativeSQLAdmission{test: t, side: "hub", origins: f.origins, peer: peer, router: f.router, leaf: certificate.Certificate[0], connectorVerifier: connectorVerifier, connectorDigest: sha256.Sum256(connectorCertificate.Certificate[0])}
	upstream := nativeSQLServer(t, contracts.NativeOwnerPath, hub, connectorCertificate, keyTLS)
	connector := &nativeSQLAdmission{test: t, side: "connector", origins: f.origins, peer: peer, upstream: upstream, verifier: f.enrollment.issuer.verifier}
	client := nativeSQLServer(t, contracts.NativeWorkerPath, connector, certificate, keyTLS)
	call, cancel := context.WithDeadline(ctx, f.target.RunUntil)
	done := make(chan error, 1)
	joined := make(chan nativeSQLWorkerOutcome, 3)
	f.joined = joined
	go func() { done <- nativeSQLWorkerLoop(t, call, client, f.target, digest, result, joined) }()
	var once sync.Once
	f.closeWorker = func() {
		once.Do(func() {
			cancel()
			err := <-done
			if err != nil && !errors.Is(err, context.Canceled) {
				t.Error("native acquisition terminated", err)
			}
		})
	}
	t.Cleanup(f.closeWorker)
}
