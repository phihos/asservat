// SQL admission, role isolation, encrypted pair writes, current recovery and
// exact leaf activation are real. Kubernetes/source/qualification facts and the
// protected issuer are explicitly synthetic. This is not an installed connector
// or qualification of production backup cost, Pod proof or OpenBao issuance.
type connectorSQLFixture struct {
	t                        *testing.T
	ctx                      context.Context
	admin                    *pgxpool.Pool
	pools                    map[string]*pgxpool.Pool
	scenario                 captureAdmissionScenario
	configuration            app.ConnectorIdentityConfiguration
	source                   *postgres.ConnectorIdentitySources
	observer                 *postgres.FleetObservation
	processes                *postgres.ConnectorProcesses
	credentials              *postgres.ConnectorCredentials
	backing                  *postgres.ConnectorBackingJournal

	mac                      *auditapp.OperationMAC
	enrollment               app.ConnectorProcessEnrollment
	renewal                  app.ConnectorRenewal
	request                  value.ConnectorProcessRequest
	processID                string
	process                  value.ConnectorCredentialProcess
	clientCA, serverCA       *x509.Certificate
	clientCAKey, serverCAKey *ecdsa.PrivateKey
}

func connectorSQLCA(t *testing.T) (*x509.Certificate, *ecdsa.PrivateKey) {
	t.Helper()
	key, e := ecdsa.GenerateKey(elliptic.P256(), rand.Reader)
	if e != nil {
		t.Fatal(e)
	}
	n := time.Now().UTC().Truncate(time.Second)
	c := &x509.Certificate{SerialNumber: big.NewInt(n.UnixNano()), IsCA: true, BasicConstraintsValid: true, KeyUsage: x509.KeyUsageCertSign | x509.KeyUsageCRLSign, NotBefore: n.Add(-time.Minute), NotAfter: n.Add(24 * time.Hour)}
	der, e := x509.CreateCertificate(rand.Reader, c, c, key.Public(), key)
	if e != nil {
		t.Fatal(e)
	}
	parsed, e := x509.ParseCertificate(der)
	if e != nil {
		t.Fatal(e)
	}
	return parsed, key
}
func connectorSQLCSR(t *testing.T) []byte {
	t.Helper()
	k, e := ecdsa.GenerateKey(elliptic.P256(), rand.Reader)
	if e != nil {
		t.Fatal(e)
	}
	b, e := x509.CreateCertificateRequest(rand.Reader, &x509.CertificateRequest{}, k)
	if e != nil {
		t.Fatal(e)
	}
	return b
}

func (f *connectorSQLFixture) refresh(t *testing.T) {
	t.Helper()
	r, e := f.scenario.recovery.reader.RecoveryState(f.ctx)
	if e != nil { // Read original accepted issuer; observation never activates or clears recovery.
		var epoch string
		if e = f.admin.QueryRow(f.ctx, "SELECT issuer_epoch::text FROM fleet_trust WHERE installation_id=$1", f.scenario.recovery.installation).Scan(&epoch); e != nil {
			t.Fatal(e)
		}
		r.IssuerEpoch = epoch
	}
	n := time.Now().UTC().Truncate(time.Microsecond)
	var digest artifact.Digest
	copy(digest[:], bytes.Repeat([]byte{97}, 32))
	a := value.Administration{InstallationID: f.scenario.recovery.installation, IssuerEpoch: r.IssuerEpoch, Issuer: "https://identity.invalid", Revision: identity.Revision{UID: "synthetic-recovery", Generation: 1, Digest: digest}, ObservedAt: n, FreshUntil: n.Add(5 * time.Second), Clients: []string{"console"}, DeliveryProfiles: []string{"synthetic"}, MaxActiveBootstraps: 1, MaxBootstrapHistory: 16, MaxClusters: 2}
	if e = f.observer.ObserveAdministration(f.ctx, a); e != nil {
		t.Fatal("refresh existing synthetic fleet observation", e)
	}
}
func (f *connectorSQLFixture) ReadConnectorIdentityConfiguration(ctx context.Context) (app.ConnectorIdentityConfiguration, error) {
	c := f.configuration.Clone()
	n := time.Now().UTC().Truncate(time.Microsecond)
	c.Source.Observation.ObservedAt = n
	c.Source.Observation.FreshUntil = n.Add(5 * time.Second)
	if d, ok := ctx.Deadline(); ok && d.Before(c.Source.Observation.FreshUntil) {
		c.Source.Observation.FreshUntil = d.UTC().Truncate(time.Microsecond)
	}
	return c, ctx.Err()
}

func (f *connectorSQLFixture) SignConnectorCredential(ctx context.Context, p value.ConnectorCredentialPermit, csr []byte) (value.IssuedConnectorCredential, error) {
	if ctx.Err() != nil || !p.Valid(time.Now()) {
		return value.IssuedConnectorCredential{}, value.ErrDenied
	}
	q, e := x509.ParseCertificateRequest(csr)
	if e != nil {
		return value.IssuedConnectorCredential{}, e
	}
	ca, key, usage := f.clientCA, f.clientCAKey, x509.ExtKeyUsageClientAuth
	name := ""
	if p.Usage == "server" {
		ca, key, usage, name = f.serverCA, f.serverCAKey, x509.ExtKeyUsageServerAuth, p.Configuration.Profile.ServerName
	}
	uri, _ := mtls.PeerURI(p.Process.Binding)
	n := p.CheckedAt.UTC().Truncate(time.Second)
	leaf := &x509.Certificate{SerialNumber: big.NewInt(time.Now().UnixNano()), BasicConstraintsValid: true, KeyUsage: x509.KeyUsageDigitalSignature, ExtKeyUsage: []x509.ExtKeyUsage{usage}, URIs: []*url.URL{uri}, NotBefore: n.Add(-time.Second), NotAfter: p.CertificateNotAfter}
	if name != "" {
		leaf.DNSNames = []string{name}
	}
	der, e := x509.CreateCertificate(rand.Reader, leaf, ca, q.PublicKey, key)
	if e != nil {
		return value.IssuedConnectorCredential{}, e
	}
	return value.IssuedConnectorCredential{Certificate: value.ConnectorCredentialCertificate{Binding: p.Process.Binding, Usage: p.Usage, ServerName: name, Digest: artifact.Digest(sha256.Sum256(der)), PublicKeyDigest: p.PublicKeyDigest(), NotBefore: leaf.NotBefore, NotAfter: leaf.NotAfter}, DER: der, ChainDER: [][]byte{ca.Raw}}, nil
}

func (f *connectorSQLFixture) renewBacking(t *testing.T, promote ...bool) string {
	t.Helper()
	f.refresh(t)
	before := f.configuration.Backing.Clone()
	f.configuration.Backing.Policy.BackingRevision++
	f.configuration.Backing.Policy.ValidUntil = f.configuration.Backing.Policy.ValidUntil.Add(10 * time.Minute)
	f.configuration.Backing.Policy.PolicyID = custodyID()
	f.configuration.Backing.Policy.PolicyDigest[0]++
	f.configuration.Source.Observation.Backing.Revision++
	f.configuration.Source.Observation.Backing.Digest[0]++
	f.configuration.Source.Observation.Backing.SourceDigest[0]++
	a, _ := json.Marshal(before)
	b, _ := json.Marshal(f.configuration.Backing)
	var compatible bool
	if e := f.admin.QueryRow(f.ctx, "SELECT connector_backing_can_renew($1::jsonb,$2::jsonb)", a, b).Scan(&compatible); e != nil || !compatible {
		t.Fatal("actual SQL backing compatibility", compatible, e)
	}
	c, _ := f.ReadConnectorIdentityConfiguration(f.ctx)
	if e := f.source.StageConnectorIdentity(f.ctx, c); e != nil {
		t.Fatal("stage backing", e)
	}
	proof, e := f.ReadConnectorProcessProof(f.ctx, f.request)
	if e != nil {
		t.Fatal(e)
	}
	r, e := f.scenario.recovery.reader.RecoveryState(f.ctx)
	if e != nil {
		t.Fatal(e)
	}
	n := time.Now().UTC().Truncate(time.Microsecond)
	until := identity.Earlier(proof.FreshUntil, identity.Earlier(c.Source.Observation.FreshUntil, r.FreshUntil))
	current := app.ConnectorBackingCurrent{Configuration: c, Proof: proof, Recovery: r, ObservedAt: n, Until: until}
	id, e := f.backing.ConnectorBackingCandidateID(f.ctx, current)
	if e != nil {
		t.Fatal("candidate lookup", e)
	}
	if len(promote) > 0 && !promote[0] {
		return id
	}
	service := app.ConnectorBackingAcceptance{Source: f, Proof: f, Recovery: f.scenario.recovery.reader, Journal: f.backing, Promotion: f.source, Audit: f.audit, MAC: f.mac, Now: time.Now}
	if _, e = service.Accept(f.ctx, id, f.request); e != nil {
		t.Fatal("backing acceptance", e)
	}
	return id
}
func qualifyConnectorIdentity(t *testing.T, ctx context.Context, admin *pgxpool.Pool, poolFor func(string, string) *pgxpool.Pool, store *objectstore.Store, keys authorityTestKeys, create func(*testing.T, int64) captureAdmissionScenario) {
	t.Run("connector-identity-fast", func(t *testing.T) {
		f := newConnectorSQLFixture(t, ctx, admin, poolFor, store, keys, create(t, 1), 10800)
		client, server := f.issue(t, "client"), f.issue(t, "server")
		for _, leaf := range []value.IssuedConnectorCredential{client, server} {
			if _, e := f.registry.ConnectorLeafRegistration(ctx, leaf.Certificate); e != nil {
				t.Fatal("exact active leaf", e)
			}
		}
		var oldProcess []byte
		if e := admin.QueryRow(ctx, "SELECT process::text FROM connector_identity_process WHERE id=$1", f.processID).Scan(&oldProcess); e != nil {
			t.Fatal(e)
		}
		qualifyConnectorLiveProfileCleanupGuard(t, f)
		f.renewBacking(t)
		for _, leaf := range []value.IssuedConnectorCredential{client, server} {
			if _, e := f.registry.ConnectorLeafRegistration(ctx, leaf.Certificate); e != nil {
				t.Fatal("backing renewal invalidated original live leaf", e)
			}
		}
		// The second permitted provider attempt exercises a new exact paid intent
		// under the new policy without waiting for the next 30-minute signing period.
		replacement := f.issue(t, "client")
		if replacement.Certificate.Digest == client.Certificate.Digest {
			t.Fatal("new exact attempt did not create independent leaf")
		}
		var retained []byte
		if e := admin.QueryRow(ctx, "SELECT process::text FROM connector_identity_process WHERE id=$1", f.processID).Scan(&retained); e != nil || !bytes.Equal(oldProcess, retained) {
			t.Fatal("backing renewal rewrote original process/dates", e)
		}
		for role, p := range f.pools {
			for _, table := range []string{"operation_audit", "operation_audit_account", "connector_identity_root", "connector_identity_process"} {
				if _, e := p.Exec(ctx, "SELECT * FROM "+table+" LIMIT 1"); e == nil {
					t.Fatal("private role has ambient base read", role, table)
				}
			}
		}

		for _, mode := range []string{"close", "promote"} {
			t.Run("exhausted-source-"+mode, func(t *testing.T) {
				bounded := newConnectorSQLFixture(t, ctx, admin, poolFor, store, keys, create(t, 1), 10800, 4)
				candidate := bounded.renewBacking(t, mode == "promote")
				var used, limit int64
				if e := admin.QueryRow(ctx, "SELECT source_transitions,source_transition_limit FROM connector_identity_root WHERE root_id=$1", bounded.scenario.a.RootID).Scan(&used, &limit); e != nil || used != limit {
					t.Fatal("fixture did not exhaust source backing", used, limit, e)
				}
				if mode == "close" {
					var closed bool
					if e := bounded.pools["source"].QueryRow(ctx, "SELECT close_connector_backing_candidate($1,$2)", bounded.scenario.a.RootID, candidate).Scan(&closed); e != nil || !closed {
						t.Fatal("prepaid exact candidate close denied", e)
					}
					if e := bounded.pools["source"].QueryRow(ctx, "SELECT close_connector_backing_candidate($1,$2)", bounded.scenario.a.RootID, candidate).Scan(&closed); e != nil || !closed {
						t.Fatal("same closed candidate replay denied", e)
					}
				}
				if _, e := bounded.enrollment.Enroll(ctx, custodyID(), bounded.request); e == nil {
					t.Fatal("exhausted negative-watermark budget admitted serving authority")
				}
				original := bounded.configuration.Source.Observation.Backing.SourceDigest
				bounded.configuration.Source.Observation.Backing.SourceDigest[0]++
				changed, _ := bounded.ReadConnectorIdentityConfiguration(ctx)
				if e := bounded.source.StageConnectorIdentity(ctx, changed); e == nil {
					t.Fatal("contrary exhausted source accepted")
				}
				bounded.configuration.Source.Observation.Backing.SourceDigest = original
				restored, _ := bounded.ReadConnectorIdentityConfiguration(ctx)
				_ = bounded.source.StageConnectorIdentity(ctx, restored)
				if _, e := bounded.enrollment.Enroll(ctx, custodyID(), bounded.request); e == nil {
					t.Fatal("restoring old source revived exhausted authority")
				}
				var final int64
				if e := admin.QueryRow(ctx, "SELECT source_transitions FROM connector_identity_root WHERE root_id=$1", bounded.scenario.a.RootID).Scan(&final); e != nil || final != used {
					t.Fatal("resolution refilled/spent unreserved transition", final, used, e)
				}
			})
		}

		f.qualifyPrivateViews(t, poolFor)
		t.Log("real SQL/S3 admitted process, distinct client/server activation, backing continuation and original dates passed; source/issuer facts synthetic")
	})
}

func (f *connectorSQLFixture) qualifyPrivateViews(t *testing.T, poolFor func(string, string) *pgxpool.Pool) {
	t.Helper()
	parentPool := poolFor
	poolFor = func(role, password string) *pgxpool.Pool {
		p := parentPool(role, password)
		t.Cleanup(p.Close)
		return p
	}
	f.refresh(t)
	ctx := f.ctx
	root := f.scenario.a.RootID
	normal, e := postgres.New(ctx, poolFor("operation_audit_config_runtime", "synthetic-operation-config"))
	if e != nil {
		t.Fatal(e)
	}
	n := time.Now().UTC().Truncate(time.Microsecond)
	if e = normal.ProvisionOperationAuditPool(ctx, auditapp.OperationAuditPool{RootID: root, Lane: "normal", AccountVersion: "synthetic-human", Accounts: 1, RecordsPerAccount: 2, RetainedSeconds: 3600, ObservedAt: n, FreshUntil: n.Add(5 * time.Second), ValidUntil: n.Add(time.Hour)}); e != nil {
		t.Fatal(e)
	}
	j, e := postgres.NewOperationJournal(ctx, poolFor("operation_audit_runtime", "synthetic-operation-journal"), root, "normal", f.scenario.recovery.guard)
	if e != nil {
		t.Fatal(e)
	}
	human := audit.OperationIntent{ID: custodyID(), RootID: root, TraceID: custodyID(), Name: "get_case", InputFormat: "context-input/v1", MACVersion: "synthetic", AccountVersion: "synthetic-human", Actor: identity.Principal{Issuer: "https://identity.invalid", Subject: "human-secret", ClientID: "console", Kind: identity.Human}, PrincipalDigest: artifact.Digest{41}, InputDigest: artifact.Digest{42}, CreatedAt: n, RequestUntil: n.Add(5 * time.Second), WriteUntil: n.Add(30 * time.Second), RetainUntil: n.Add(time.Hour)}
	if _, e = j.ReserveOperation(ctx, human); e != nil {
		t.Fatal("positive ordinary row for isolation", e)
	}
	for _, duty := range []string{"cleanup", "verifier", "finalizer"} {
		p := f.pools[duty]
		for _, view := range []string{"pool", "account", "audit", "lifecycle", "history"} {
			var wrong int64
			if e = p.QueryRow(ctx, "SELECT count(*) FROM connector_operation_"+view+" WHERE lane<>'identity'").Scan(&wrong); e != nil || wrong != 0 {
				t.Fatal("human row visible through filtered view", duty, view, wrong, e)
			}
		}
		var own int64
		if e = p.QueryRow(ctx, "SELECT count(*) FROM connector_operation_audit WHERE root_id=$1 AND id=$2", root, f.processID).Scan(&own); e != nil || own != 1 {
			t.Fatal("private positive view not exercised", duty, own, e)
		}
		var humanCount int64
		if e = p.QueryRow(ctx, "SELECT count(*) FROM connector_operation_audit WHERE id=$1", human.ID).Scan(&humanCount); e != nil || humanCount != 0 {
			t.Fatal("known human ID disclosed", duty, e)
		}
		tag, e := p.Exec(ctx, "UPDATE connector_operation_audit SET id=id WHERE id=$1", human.ID)
		if e != nil || tag.RowsAffected() != 0 {
			t.Fatal("human immutable-key UPDATE reached a row", duty, e)
		}
		for _, sql := range []string{
			"SELECT (SELECT intent FROM operation_audit LIMIT 1) FROM connector_operation_audit",
			"UPDATE connector_operation_audit SET id='20000000-0000-4000-8000-000000000001' WHERE id='" + f.processID + "' RETURNING intent",
			"UPDATE connector_operation_lifecycle SET intent_completion='{}' WHERE operation_id='" + f.processID + "'",
			"INSERT INTO connector_operation_audit SELECT * FROM connector_operation_audit RETURNING intent",
			"DELETE FROM connector_operation_audit RETURNING intent",
			"SET ROLE postgres",
		} {
			if _, e = p.Exec(ctx, sql); e == nil {
				t.Fatal("private role escaped its exact duty", duty, sql)
			}
		}
	}
	// A view owned by the migration owner still evaluates current_user as the
	// invoking function owner, which is deliberately different from session_user.
	fn := "connector_view_probe_" + strings.ReplaceAll(custodyID(), "-", "")
	if _, e = f.admin.Exec(ctx, "CREATE FUNCTION "+pgx.Identifier{fn}.Sanitize()+"() RETURNS bigint LANGUAGE sql SECURITY DEFINER SET search_path=pg_catalog,pg_temp AS 'SELECT count(*) FROM public.connector_operation_audit'"); e != nil {
		t.Fatal(e)
	}
	defer f.admin.Exec(ctx, "DROP FUNCTION "+pgx.Identifier{fn}.Sanitize()+"()")
	var throughOwner int64
	if e = f.pools["cleanup"].QueryRow(ctx, "SELECT "+pgx.Identifier{fn}.Sanitize()+"()").Scan(&throughOwner); e != nil || throughOwner != 0 {
		t.Fatal("view owner turned private login into ambient reader", throughOwner, e)
	}
	// Bounded waiting must preserve the caller's original deadline, even when
	// SELECT FOR UPDATE is permitted on an otherwise immutable key column.
	tx, e := f.admin.Begin(ctx)
	if e != nil {
		t.Fatal(e)
	}
	defer tx.Rollback(ctx)
	if _, e = tx.Exec(ctx, "SELECT id FROM operation_audit WHERE id=$1 FOR UPDATE", f.processID); e != nil {
		t.Fatal(e)
	}
	wait, stop := context.WithTimeout(ctx, 100*time.Millisecond)
	defer stop()
	began := time.Now()
	_, e = f.pools["cleanup"].Exec(wait, "SELECT id FROM connector_operation_audit WHERE id=$1 FOR UPDATE", f.processID)
	if e == nil || time.Since(began) > time.Second {
		t.Fatal("private view ignored original lock deadline", e, time.Since(began))
	}
}
