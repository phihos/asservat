package postgres

import (
	"context"
	"crypto/tls"
	"sync"
	"time"

	app "asservat.local/asservat/internal/application/fleet"
	"asservat.local/asservat/internal/domain/audit"
	cfg "asservat.local/asservat/internal/domain/configuration"
	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgtype"
	"github.com/jackc/pgx/v5/pgxpool"
)

// InitialRuntimeBridgeSQL is a separate post-activation administrator facet.
// No credential from the retired initial manager or activation Pod is revived.
// Stored observation freshness is not a substitute for the native installer
// proof, and this facet never writes fleet activation, recovery or source state.
type InitialRuntimeBridgeSQL struct {
	pool          *pgxpool.Pool
	record        audit.InitialInstallation
	receipt       audit.Receipt
	administrator uint32
	ctx           context.Context
	cancel        context.CancelFunc
	mu            sync.Mutex
	closed, busy  bool
	joined        sync.WaitGroup
}

var _ app.InitialRuntimeBridgeEffects = (*InitialRuntimeBridgeSQL)(nil)

func NewInitialRuntimeBridgeSQL(ctx context.Context, config *pgx.ConnConfig, record audit.InitialInstallation, receipt audit.Receipt) (*InitialRuntimeBridgeSQL, error) {
	if ctx == nil || ctx.Err() != nil || !record.Valid() || record.Claim.Database.SchemaVersion != currentSchemaVersion || !receipt.ValidInitialInstallation(record.Scope.InstallationID, record.Claim.ID) || config == nil || config.User == "" || config.Database == "" || config.TLSConfig == nil || config.TLSConfig.InsecureSkipVerify || config.TLSConfig.ServerName == "" || config.TLSConfig.MinVersion < tls.VersionTLS13 || len(config.Fallbacks) != 0 || config.Tracer != nil || config.ValidateConnect != nil || config.AfterConnect != nil {
		return nil, cfg.ErrInvalid
	}
	config = config.Copy()
	config.RuntimeParams = map[string]string{"application_name": "asservat-runtime-guard-bridge", "search_path": "pg_catalog,public", "timezone": "UTC", "lock_timeout": "1000ms", "statement_timeout": "5000ms", "idle_in_transaction_session_timeout": "5000ms", "synchronous_commit": "on", "row_security": "off"}
	config.ConnectTimeout = 5 * time.Second
	p, err := pgxpool.ParseConfig("")
	if err != nil {
		return nil, cfg.ErrInvalid
	}
	p.ConnConfig, p.MaxConns, p.MinConns = config, 1, 0
	life, cancel := context.WithCancel(ctx)
	pool, err := pgxpool.NewWithConfig(life, p)
	if err != nil {
		cancel()
		return nil, cfg.ErrUnavailable
	}
	out := &InitialRuntimeBridgeSQL{pool: pool, record: record, receipt: receipt, ctx: life, cancel: cancel}
	call, end := context.WithTimeout(life, 5*time.Second)
	defer end()
	var valid bool
	err = pool.QueryRow(call, `SELECT oid,rolsuper AND rolcanlogin AND current_user=session_user AND NOT pg_is_in_recovery() FROM pg_roles WHERE rolname=session_user`).Scan(&out.administrator, &valid)
	if err != nil || !valid {
		out.Close()
		return nil, cfg.ErrDenied
	}
	// This administrator-only installer must not use the runtime Ledger
	// constructor, which correctly denies administrator credentials. Validate
	// this purpose's schema and original incarnation without creating any
	// runtime facet or relaxing the Ledger's independent role restriction.
	until, _ := call.Deadline()
	if err = out.current(call, pool, until); err != nil {
		out.Close()
		return nil, err
	}
	// Construction is read-only. Each actual effect reacquires all locks and
	// checks the exact record independently; construction grants no cached permit.
	if err = out.WithInitialRuntimeBridge(call, func(context.Context) error { return nil }); err != nil {
		out.Close()
		return nil, err
	}
	return out, nil
}

func (s *InitialRuntimeBridgeSQL) InitialRuntimeBridgeRecord() (audit.InitialInstallation, audit.Receipt) {
	if s == nil {
		return audit.InitialInstallation{}, audit.Receipt{}
	}
	return s.record, s.receipt
}
func (s *InitialRuntimeBridgeSQL) Close() {
	if s == nil {
		return
	}
	s.mu.Lock()
	s.closed = true
	s.cancel()
	s.mu.Unlock()
	s.joined.Wait()
	s.pool.Close()
}
func (s *InitialRuntimeBridgeSQL) WithInitialRuntimeBridge(ctx context.Context, effect func(context.Context) error) error {
	if s == nil || ctx == nil || ctx.Err() != nil || effect == nil {
		return cfg.ErrInvalid
	}
	s.mu.Lock()
	if s.closed || s.busy || s.ctx.Err() != nil {
		s.mu.Unlock()
		return cfg.ErrUnavailable
	}
	s.busy = true
	s.joined.Add(1)
	s.mu.Unlock()
	// This start precedes pool acquisition and both blocking row locks. Nothing
	// after a wait can renew it. A shorter native proof narrows the child context.
	call, cancel := context.WithTimeout(ctx, 5*time.Second)
	done := make(chan struct{})
	stop := context.AfterFunc(s.ctx, func() { cancel(); close(done) })
	defer func() {
		cancel()
		if !stop() {
			<-done
		}
		s.mu.Lock()
		s.busy = false
		s.mu.Unlock()
		s.joined.Done()
	}()
	until, _ := call.Deadline()
	tx, err := s.pool.BeginTx(call, pgx.TxOptions{})
	if err != nil {
		return cfg.ErrUnavailable
	}
	defer func() {
		cleanup, end := context.WithTimeout(context.Background(), time.Second)
		defer end()
		_ = tx.Rollback(cleanup)
	}()
	// Existing global lock order: current recovery, then original activation.
	// FOR SHARE blocks a concurrent suspension/incarnation acceptance until all
	// native I/O and its result check have joined this operation.
	var valid bool
	err = tx.QueryRow(call, `SELECT NOT recovery_blocked AND NOT suspended AND issuer_epoch=$2 AND source_uid=$3
 FROM public.fleet_current_trust WHERE installation_id=$1 FOR SHARE`, s.record.Scope.InstallationID, s.record.Scope.IssuerEpoch, s.record.Claim.FleetSource.UID).Scan(&valid)
	if err != nil || !valid {
		return cfg.ErrUnavailable
	}
	var raw, receipt []byte
	var at pgtype.Timestamptz
	var claim pgtype.UUID
	err = tx.QueryRow(call, `SELECT record_json,audit_receipt_json,activated_at,claim_id FROM public.fleet_initial_activation WHERE installation_id=$1 FOR SHARE`, s.record.Scope.InstallationID).Scan(&raw, &receipt, &at, &claim)
	if err != nil {
		return cfg.ErrUnavailable
	}
	observed, err := initialInstallationObservation(raw, receipt, at, s.record.Scope, s.record.Claim.PlanDigest)
	if err != nil || observed.Disposition != app.InitialInstallationActivated || observed.Record == nil || !observed.Record.Claim.Same(s.record.Claim) || observed.AuditReceipt == nil || *observed.AuditReceipt != s.receipt || uid(claim) != s.record.Claim.ID {
		return cfg.ErrChanged
	}
	if err = s.current(call, tx, until); err != nil {
		return err
	}
	if err = effect(call); err != nil {
		return err
	}
	// A late/unknown native outcome stays unknown. SQL does not copy a receipt
	// or convert elapsed time into successful completion.
	return s.current(call, tx, until)
}
func (s *InitialRuntimeBridgeSQL) current(ctx context.Context, q publicRoleQuery, until time.Time) error {
	var valid bool
	err := q.QueryRow(ctx, `SELECT p.oid=$1 AND p.rolsuper AND p.rolcanlogin AND current_user=session_user
 AND NOT pg_is_in_recovery() AND pg_postmaster_start_time()=$2
 AND (SELECT oid FROM pg_database WHERE datname=current_database())=$3
 AND clock_timestamp() < $4
 AND COALESCE((SELECT version_id=$5 AND is_applied FROM public.goose_db_version ORDER BY id DESC LIMIT 1),false)
 FROM pg_roles p WHERE p.rolname=session_user`, s.administrator, s.record.Claim.Database.StartedAt, s.record.Claim.Database.DatabaseOID, until, currentSchemaVersion).Scan(&valid)
	if err != nil || !valid || ctx.Err() != nil || !time.Now().Before(until) {
		return cfg.ErrUnavailable
	}
	return nil
}
