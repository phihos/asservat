// This helper uses real h2/mTLS and process-owned keys. Native source, workload
// and SQL registry values are the explicitly synthetic channel fixture; it is
// not installed delegation or two-cluster availability qualification.
func operationStreamClient(t *testing.T, x *observationChannelFixture, handler http.HandlerFunc, maximum int, changeTLS ...func(*tls.Config)) *ConnectorObservationOperationClient {
	t.Helper()
	server := httptest.NewUnstartedServer(handler)
	_, port, err := net.SplitHostPort(server.Listener.Addr().String())
	if err != nil {
		t.Fatal(err)
	}
	x.ownerPin.Pool.NativePort, err = strconv.Atoi(port)
	if err != nil {
		t.Fatal(err)
	}
	common, _ := OwnerBootstrapServerName(x.ownerPin.Pool)
	certificate := x.ownerCA.leaf(t, x.f.clock(), ownerBootstrapBinding(ownerTestID), true, func(c *x509.Certificate) { c.DNSNames = append(c.DNSNames, common) })
	parsed, err := x509.ParseCertificate(certificate.Certificate[0])
	if err != nil {
		t.Fatal(err)
	}
	verifier, err := NewOwnerPeerVerifierForPool(x.ownerCA.roots, x.ownerPin.Pool, x.f.clock)
	if err != nil {
		t.Fatal(err)
	}
	x.owner, err = verifier.verifyCertificates([]*x509.Certificate{parsed}, "server")
	if err != nil {
		t.Fatal(err)
	}
	server.EnableHTTP2 = true
	server.TLS = &tls.Config{Certificates: []tls.Certificate{certificate}, ClientAuth: tls.RequireAndVerifyClientCert, ClientCAs: x.f.clientCA.roots, MinVersion: tls.VersionTLS13, MaxVersion: tls.VersionTLS13, NextProtos: []string{"h2"}, SessionTicketsDisabled: true}
	for _, change := range changeTLS {
		change(server.TLS)
	}
	server.StartTLS()
	candidates := observationCandidatesFunc(func(context.Context) (value.APIContextOwnerCandidates, error) {
		x.candidateCalls++
		owners := []value.OwnerBinding{x.owner.Binding}
		if x.missing {
			owners = nil
		}
		return value.APIContextOwnerCandidates{Pool: x.ownerPin.Pool, Owners: owners, ObservedAt: x.f.clock(), ValidUntil: x.f.clock().Add(5 * time.Second)}, nil
	})
	registry := forwardRegistry(func(context.Context, value.OwnerCertificate) (value.OwnerRegistrationObservation, error) {
		x.registryCalls++
		if x.revoked {
			return value.OwnerRegistrationObservation{}, value.ErrUnavailable
		}
		_, until, _ := value.OwnerCertificateWindow(x.f.now)
		enrollment := value.OwnerEnrollment{ID: ownerTestID, Binding: x.owner.Binding, Usage: "server", Pool: x.ownerPin.Pool, SourceUID: x.ownerPin.SourceUID, Pod: "owner-1", ReplicaSet: "owner-rs", ReplicaSetUID: ownerTestID, PublicKeyDigest: sha256.Sum256(parsed.RawSubjectPublicKeyInfo), InitialTokenDigest: artifact.Digest{1}, InitialWorkloadDigest: artifact.Digest{2}, InitialProtectionDigest: x.ownerPin.ProtectionDigest, ClaimedAt: x.f.now, InitialHoldUntil: until.Add(value.OwnerRegistrationGrace)}
		return value.OwnerRegistrationObservation{Enrollment: enrollment, Certificate: x.owner, Configuration: ownerBootstrapConfiguration(x.ownerPin.Pool, x.ownerCA, x.f.clock()), ObservedAt: x.f.clock(), FreshUntil: x.f.clock().Add(5 * time.Second)}, nil
	})
	producer, err := NewConnectorObservationOperationProducer(x.channel.producer.pin, x.channel.producer.credentials, &connectorOperationNativeFixture{f: x.f, pin: x.channel.producer.pin}, x.f.clock)
	if err != nil {
		server.Close()
		t.Fatal(err)
	}
	client, err := NewConnectorObservationOperationClient(context.Background(), x.ownerPin, server.Listener.Addr().String(), x.ownerCA.roots, producer, candidates, registry, x.channel.metadata, maximum, x.f.clock)
	if err != nil {
		server.Close()
		t.Fatal(err)
	}
	t.Cleanup(func() { client.Close(); server.Close() })
	return client
}

func operationStreamHello(w http.ResponseWriter, r *http.Request) (contracts.ClusterObservationHello, error) {
	if r.ProtoMajor != 2 || r.TLS == nil || !r.TLS.HandshakeComplete || r.ContentLength != -1 || r.Method != http.MethodPost || r.URL.Path != contracts.ClusterObservationOperationExchangePath || r.Header.Get("Authorization") != "" || r.Header.Get("Cookie") != "" || r.Header.Get("Idempotency-Key") != "" {
		return contracts.ClusterObservationHello{}, value.ErrDenied
	}
	frame, err := contracts.ReadClusterObservationFrame(r.Body)
	if err != nil || frame.Kind != contracts.ClusterObservationHelloFrame {
		return contracts.ClusterObservationHello{}, value.ErrDenied
	}
	hello, err := contracts.DecodeClusterObservationHello(frame.Payload)
	clear(frame.Payload)
	if err != nil {
		return hello, err
	}
	w.Header().Set("Content-Type", contracts.ClusterObservationOperationExchangeMedia)
	w.Header().Set("Cache-Control", "no-store")
	w.WriteHeader(http.StatusOK)
	err = http.NewResponseController(w).Flush()
	return hello, err
}

func TestConnectorOperationStreamStandbyCloseJoinsBlockedReadAndRetainsSlot(t *testing.T) {
	x := newObservationChannelFixture(t)
	var calls atomic.Int32
	serverJoined := make(chan struct{})
	client := operationStreamClient(t, x, func(w http.ResponseWriter, r *http.Request) {
		calls.Add(1)
		defer close(serverJoined)
		if _, err := operationStreamHello(w, r); err != nil {
			return
		}
		<-r.Context().Done()
	}, 1)
	ctx, cancel := context.WithTimeout(context.Background(), 4*time.Second)
	defer cancel()
	stream, err := client.Open(ctx, ownerOtherID)
	if err != nil {
		t.Fatal(err)
	}
	if x.registryCalls != 0 || x.metadataCalls != 0 {
		t.Fatal("empty standby performed business proof/read")
	}
	if _, err = client.Open(ctx, ownerTestID); err == nil || calls.Load() != 1 {
		t.Fatal("occupied standby slot reused")
	}
	read := make(chan error, 1)
	entered := make(chan struct{})
	stream.response = &observationEnteredRead{ReadCloser: stream.response, entered: entered}
	go func() { _, e := stream.Receive(ctx); read <- e }()
	select {
	case <-entered:
	case <-time.After(time.Second):
		t.Fatal("read never reached actual body")
	}
	// Close waits for the actual blocked Receive; it cannot be merely a flag.
	if err = stream.Close(); err != nil {
		t.Fatal(err)
	}
	select {
	case e := <-read:
		if e == nil {
			t.Fatal("cancelled read succeeded")
		}
	case <-time.After(time.Second):
		t.Fatal("Close did not join admitted read")
	}
	select {
	case <-serverJoined:
	case <-time.After(time.Second):
		t.Fatal("actual HTTP handler did not observe stream closure")
	}
	client.base.mu.Lock()
	active := client.base.active
	client.base.mu.Unlock()
	if active != 0 || calls.Load() != 1 {
		t.Fatal("closed stream retained slot or retried", active, calls.Load())
	}
}

func TestConnectorOperationStreamUnknownPOSTIsNotReplayed(t *testing.T) {
	for _, mode := range []string{"unavailable", "redirect", "fixed-length", "wrong-media"} {
		t.Run(mode, func(t *testing.T) {
			x := newObservationChannelFixture(t)
			var calls atomic.Int32
			client := operationStreamClient(t, x, func(w http.ResponseWriter, r *http.Request) {
				calls.Add(1)
				_, _ = contracts.ReadClusterObservationFrame(r.Body)
				w.Header().Set("Content-Type", contracts.ClusterObservationOperationExchangeMedia)
				w.Header().Set("Cache-Control", "no-store")
				switch mode {
				case "unavailable":
					w.WriteHeader(http.StatusServiceUnavailable)
				case "redirect":
					w.Header().Set("Location", contracts.ClusterObservationOperationExchangePath)
					w.WriteHeader(http.StatusTemporaryRedirect)
				case "fixed-length":
					w.Header().Set("Content-Length", "0")
					w.WriteHeader(http.StatusOK)
				case "wrong-media":
					w.Header().Set("Content-Type", "application/json")
					w.WriteHeader(http.StatusOK)
					_ = http.NewResponseController(w).Flush()
				}
			}, 1)
			ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
			defer cancel()
			if stream, err := client.Open(ctx, ownerOtherID); err == nil || stream != nil || calls.Load() != 1 || x.metadataCalls != 0 {
				t.Fatal("unknown POST admitted or retried", err, calls.Load())
			}
		})
	}
}

func TestConnectorOperationStreamRequiresActualMutualTLSAndSelectedCandidate(t *testing.T) {
	for _, mode := range []string{"wrong-name", "wrong-roots", "excluded-candidate", "no-client-certificate"} {
		t.Run(mode, func(t *testing.T) {
			x := newObservationChannelFixture(t)
			var calls atomic.Int32
			client := operationStreamClient(t, x, func(w http.ResponseWriter, r *http.Request) {
				calls.Add(1)
				w.WriteHeader(http.StatusServiceUnavailable)
			}, 1, func(c *tls.Config) {
				if mode == "no-client-certificate" {
					c.ClientAuth = tls.NoClientCert
				}
			})
			switch mode {
			case "wrong-name":
				client.base.name = "wrong.invalid"
			case "wrong-roots":
				client.base.roots = newOwnerTestCA(t, x.f.clock()).roots
			case "excluded-candidate":
				selected, err := app.NewOwnerObservationCandidateSelection(x.ownerPin.Pool, observationCandidatesFunc(func(context.Context) (value.APIContextOwnerCandidates, error) {
					return value.APIContextOwnerCandidates{Pool: x.ownerPin.Pool, Owners: []value.OwnerBinding{ownerBootstrapBinding(ownerOtherID)}, ObservedAt: x.f.clock(), ValidUntil: x.f.clock().Add(5 * time.Second)}, nil
				}))
				if err != nil {
					t.Fatal(err)
				}
				client.base.candidates = selected
			}
			ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
			defer cancel()
			if stream, err := client.Open(ctx, ownerOtherID); err == nil || stream != nil || calls.Load() != 0 || x.metadataCalls != 0 {
				t.Fatal("unqualified TLS/candidate reached Hello", mode, err, calls.Load())
			}
		})
	}
}

func TestConnectorOperationStreamOriginalDeadlineBoundsHeaderAndHello(t *testing.T) {
	for _, phase := range []string{"headers", "hello"} {
		t.Run(phase, func(t *testing.T) {
			x := newObservationChannelFixture(t)
			entered, joined := make(chan struct{}), make(chan struct{})
			client := operationStreamClient(t, x, func(w http.ResponseWriter, r *http.Request) {
				defer close(joined)
				close(entered)
				if phase == "headers" {
					_, _ = contracts.ReadClusterObservationFrame(r.Body)
				} else {
					// Reply without consuming Hello. Its original write must still
					// finish before Open can return an accepted stream.
					w.Header().Set("Content-Type", contracts.ClusterObservationOperationExchangeMedia)
					w.Header().Set("Cache-Control", "no-store")
					w.WriteHeader(http.StatusOK)
					_ = http.NewResponseController(w).Flush()
				}
				<-r.Context().Done()
			}, 1)
			ctx, cancel := context.WithTimeout(context.Background(), 400*time.Millisecond)
			defer cancel()
			stream, err := client.Open(ctx, ownerOtherID)
			if phase == "hello" && err == nil {
				// HTTP/2 flow control may positively consume this small Hello
				// before the handler reads it. The returned stream still must
				// die at the same original caller deadline.
				_, err = stream.Receive(ctx)
				_ = stream.Close()
			}
			if err == nil {
				t.Fatal("original setup/caller bound renewed")
			}
			select {
			case <-entered:
			default:
				t.Fatal("fixture failed before intended HTTP boundary")
			}
			select {
			case <-joined:
			case <-time.After(time.Second):
				t.Fatal("deadline did not join server request")
			}
		})
	}
}

func TestConnectorOperationStreamHeaderCancellationKeepsSingleShutdownOwner(t *testing.T) {
	for attempt := 0; attempt < 4; attempt++ {
		t.Run(strconv.Itoa(attempt), func(t *testing.T) {
			x := newObservationChannelFixture(t)
			ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
			defer cancel()
			serverJoined := make(chan struct{})
			client := operationStreamClient(t, x, func(w http.ResponseWriter, r *http.Request) {
				defer close(serverJoined)
				if _, err := operationStreamHello(w, r); err != nil {
					return
				}
				cancel() // exactly at the setup-to-retained-stream handoff
				<-r.Context().Done()
			}, 1)
			stream, err := client.Open(ctx, ownerOtherID)
			if err == nil {
				if _, e := stream.Receive(ctx); e == nil {
					t.Fatal("cancelled header boundary admitted a frame")
				}
				_ = stream.Close()
			}
			client.Close()
			client.base.mu.Lock()
			active := client.base.active
			client.base.mu.Unlock()
			if active != 0 || x.metadataCalls != 0 {
				t.Fatal("setup cancellation lost/double-released a slot", active)
			}
			select {
			case <-serverJoined:
			case <-time.After(time.Second):
				t.Fatal("setup cancellation did not join actual stream")
			}
		})
	}
}

func TestConnectorOperationStreamCanonicalUTCEmission(t *testing.T) {
	for _, offset := range []int{2 * 60 * 60, -7 * 60 * 60} {
		zone := time.FixedZone("fixture-offset", offset)
		f := newConnectorOperationFixture(t)
		f.request.RequestedAt = f.request.RequestedAt.In(zone)
		f.request.NotAfter = f.request.NotAfter.In(zone)
		request := observationOperationRequestWire(f.request)
		if request.RequestedAt != f.request.RequestedAt.UTC().Format(time.RFC3339Nano) || request.NotAfter != f.request.NotAfter.UTC().Format(time.RFC3339Nano) {
			t.Fatal("v2 request retained non-UTC wire text")
		}
		f.native.readHook = func(_ context.Context, _ int, v *value.ConnectorObservationNativeValue) error {
			v.NativeStartedAt = v.NativeStartedAt.In(zone)
			v.NativeUntil = v.NativeUntil.In(zone)
			v.ValueObservedAt = v.ValueObservedAt.In(zone)
			v.Source.Observation.ObservedAt = v.Source.Observation.ObservedAt.In(zone)
			v.Source.Observation.FreshUntil = v.Source.Observation.FreshUntil.In(zone)
			v.Workload.ObservedAt = v.Workload.ObservedAt.In(zone)
			v.Workload.FreshUntil = v.Workload.FreshUntil.In(zone)
			return nil
		}
		o := f.begin(t, f.root)
		w, err := o.read(f.root, 1)
		if err != nil {
			t.Fatal("regional wire", err)
		}
		assertCanonical := func(s string) {
			t.Helper()
			v, err := time.Parse(time.RFC3339Nano, s)
			if err != nil || v.UTC().Format(time.RFC3339Nano) != s {
				t.Fatal("v2 date is not canonical UTC")
			}
		}
		for _, s := range []string{w.Request.RequestedAt, w.Request.NotAfter, w.NativeStartedAt, w.NativeUntil, w.ValueObservedAt, w.SourceFreshUntil, w.WorkloadFreshUntil} {
			assertCanonical(s)
		}
		channel, err := observationOperationConnection(f.state, f.request)
		if err != nil {
			t.Fatal(err)
		}
		if _, err = parseConnectorOperationReply(w, f.pin, f.local.Registration.Certificate, f.request, 1, channel, f.f.clock()); err != nil {
			t.Fatal("same instant changed semantics", err)
		}
		x := newOperationCoordinatorFixture(t, "")
		requestValue := x.native.facts.Request
		requestValue.RequestedAt = requestValue.RequestedAt.In(zone)
		requestValue.NotAfter = requestValue.NotAfter.In(zone)
		facts := x.native.facts.Clone()
		facts.Request = requestValue
		facts.ObservedAt = facts.ObservedAt.In(zone)
		facts.FreshUntil = facts.FreshUntil.In(zone)
		leaf, err := x.server.ServingCertificate()
		if err != nil {
			t.Fatal(err)
		}
		r, err := x.registry.ReadOwnerRegistration(x.ctx, leaf)
		if err != nil {
			t.Fatal(err)
		}
		r.ObservedAt = r.ObservedAt.In(zone)
		r.FreshUntil = r.FreshUntil.In(zone)
		c := x.local.Registration.Certificate
		peer := value.PeerCertificate{Binding: c.Binding, Digest: c.Digest, NotBefore: c.NotBefore, NotAfter: c.NotAfter}
		owner, err := ownerOperationReplyWire(x.owner, requestValue, ownerTestID, 1, artifact.Digest{18}, peer, leaf, facts, r, requestValue.NotAfter, x.f.clock())
		if err != nil {
			t.Fatal("owner wire", err)
		}
		for _, s := range []string{owner.Request.RequestedAt, owner.Request.NotAfter, owner.NativeStartedAt, owner.NativeUntil, owner.RegistrationObservedAt, owner.RegistrationUntil} {
			assertCanonical(s)
		}
	}
}
