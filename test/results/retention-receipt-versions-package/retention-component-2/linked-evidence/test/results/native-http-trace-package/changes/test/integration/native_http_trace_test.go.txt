// Independent HTTP-only fixture diagnostics: original calls, contexts, return
// values, errors and marker facets are preserved. No I/O, check, retry, authority
// or goroutine is added by a decorator. Output is deferred until fixture cleanup.
// The separate recorder prevents successful preparation traffic hiding ingress.
const nativeHTTPTraceLimit = 256

type nativeHTTPTrace struct {
	mu         sync.Mutex
	start      time.Time
	events     [nativeHTTPTraceLimit]nativePreparationTraceEvent
	used       int
	omitted    uint64
	firstError *nativePreparationTraceEvent
}

func newNativeHTTPTrace(t *testing.T) *nativeHTTPTrace {
	t.Helper()
	s := &nativeHTTPTrace{start: time.Now()}
	t.Cleanup(func() { s.log(t) })
	return s
}
func (s *nativeHTTPTrace) begin(ctx context.Context, phase string, inputUntil time.Time) func(error, time.Time) {
	e := nativePreparationTraceEvent{phase: phase, started: time.Now(), inputUntil: inputUntil}
	if ctx != nil {
		e.deadline, _ = ctx.Deadline()
	}
	s.mu.Lock()
	index := -1
	if s.used < len(s.events) {
		index = s.used
		s.used++
		s.events[index] = e
	} else {
		s.omitted++
	}
	s.mu.Unlock()
	return func(err error, until time.Time) {
		e.ended, e.resultUntil, e.result = time.Now(), until, nativePreparationResult(err)
		e.contextResult = "none"
		if ctx != nil {
			e.contextResult = nativePreparationResult(ctx.Err())
		}
		s.mu.Lock()
		defer s.mu.Unlock()
		if index >= 0 {
			s.events[index] = e
		}
		if err != nil && s.firstError == nil {
			copy := e
			s.firstError = &copy
		}
	}
}
func (s *nativeHTTPTrace) log(t *testing.T) {
	t.Helper()
	s.mu.Lock()
	events := append([]nativePreparationTraceEvent(nil), s.events[:s.used]...)
	omitted := s.omitted
	var first *nativePreparationTraceEvent
	if s.firstError != nil {
		copy := *s.firstError
		first = &copy
	}
	s.mu.Unlock()
	for i, e := range events {
		if e.ended.IsZero() {
			t.Logf("native HTTP trace phase=%s seq=%d start=%s result=not-joined", e.phase, i, e.started.Sub(s.start))
			continue
		}
		t.Logf("native HTTP trace phase=%s seq=%d start=%s elapsed=%s deadline=%s remaining-start=%s remaining-end=%s input-until=%s input-remaining-end=%s result-until=%s result-remaining-end=%s result=%s context=%s", e.phase, i, e.started.Sub(s.start), e.ended.Sub(e.started), e.deadline.UTC().Format(time.RFC3339Nano), nativePreparationRemaining(e.deadline, e.started), nativePreparationRemaining(e.deadline, e.ended), e.inputUntil.UTC().Format(time.RFC3339Nano), nativePreparationRemaining(e.inputUntil, e.ended), e.resultUntil.UTC().Format(time.RFC3339Nano), nativePreparationRemaining(e.resultUntil, e.ended), e.result, e.contextResult)
	}
	if first != nil {
		t.Logf("native HTTP trace first-error phase=%s elapsed=%s result=%s context=%s", first.phase, first.ended.Sub(first.started), first.result, first.contextResult)
	}
	t.Logf("native HTTP trace retained=%d omitted=%d", len(events), omitted)
}

func (s *nativeHTTPTrace) transfer(t *testing.T, in evidence.DownloadResponseTransfer) evidence.DownloadResponseTransfer {
	t.Helper()
	original, ok := in.Responses.(nativeHTTPFirstResponses)
	if !ok {
		t.Fatal("HTTP trace requires the original first-context response facet")
	}
	in.Sessions = nativeHTTPSessions{in.Sessions, s}
	in.Responses = nativeHTTPResponses{original, s}
	in.Authority = nativeHTTPAuthority{in.Authority, s}
	in.Objects = nativeHTTPWindowObjects{in.Objects, s}
	in.Audit = nativeHTTPWindowAck{in.Audit, s}
	return in
}

// Labels remain closed even if a failing implementation supplies an unexpected
// phase. That original value still reaches the underlying port unchanged.
func nativeHTTPAuditPhase(phase, action string) string {
	switch phase {
	case "intent":
		switch action {
		case "claim":
			return "http.audit-intent-claim"
		case "ack":
			return "http.audit-intent-ack"
		case "record":
			return "http.audit-intent-record"
		case "read":
			return "http.audit-intent-read"
		}
	case "outcome":
		switch action {
		case "claim":
			return "http.audit-outcome-claim"
		case "ack":
			return "http.audit-outcome-ack"
		case "record":
			return "http.audit-outcome-record"
		case "read":
			return "http.audit-outcome-read"
		}
	}
	return "http.audit-invalid-phase"
}

// joined. It adds no effect, renewal, claim or refund. Only fixed numeric totals

func TestNativeHTTPTraceForwardsOriginalAuthority(t *testing.T) {
	ctx, cancel := context.WithDeadline(context.Background(), time.Now().Add(time.Second))
	defer cancel()
	err := errors.New("sensitive underlying error")
	until := time.Now().Add(500 * time.Millisecond)
	port := &nativeTraceTestAuthority{context: ctx, result: artifact.DownloadAuthority{ValidUntil: until}, err: err}
	trace := &nativeHTTPTrace{start: time.Now()}
	wrapper := nativeHTTPAuthority{port, trace}
	got, gotErr := wrapper.CheckDownloadCurrent(ctx, nil, artifact.DownloadScope{})
	deadline, _ := ctx.Deadline()
	if got != port.result || gotErr != err || port.calls != 1 || !port.sameContext || trace.used != 1 || trace.events[0].deadline != deadline || trace.events[0].resultUntil != until || trace.events[0].result != "other-error" {
		t.Fatal("HTTP trace changed original authority call or exposed raw failure")
	}
	cancel()
	_, gotErr = wrapper.CheckDownloadCurrent(ctx, nil, artifact.DownloadScope{})
	if gotErr != err || port.calls != 2 || trace.events[1].contextResult != "context-canceled" {
		t.Fatal("HTTP trace intercepted canceled original call")
	}
}

type nativeHTTPTestJournal struct {
	ctx      context.Context
	in       audit.OperationIntent
	out      audit.OperationOutcome
	snapshot audit.OperationSnapshot
	calls    [3]int
	same     bool
	err      error
}

func (s *nativeHTTPTestJournal) ReserveOperation(ctx context.Context, in audit.OperationIntent) (audit.OperationSnapshot, error) {
	s.calls[0]++
	s.same = s.same && ctx == s.ctx && reflect.DeepEqual(in, s.in)
	return s.snapshot, s.err
}
func (s *nativeHTTPTestJournal) ClaimOperationWrite(ctx context.Context, in audit.OperationIntent, phase string) (bool, error) {
	s.calls[1]++
	s.same = s.same && ctx == s.ctx && reflect.DeepEqual(in, s.in) && phase == "sensitive-unknown-phase"
	return false, s.err
}
func (s *nativeHTTPTestJournal) FreezeOperationOutcome(ctx context.Context, out audit.OperationOutcome) (audit.OperationSnapshot, error) {
	s.calls[2]++
	s.same = s.same && ctx == s.ctx && reflect.DeepEqual(out, s.out)
	return s.snapshot, s.err
}
func TestNativeHTTPTracePreservesAuditUnknownSpendAndSnapshot(t *testing.T) {
	ctx, cancel := context.WithCancel(context.Background())
	cancel()
	err := errors.New("private audit error")
	in := audit.OperationIntent{ID: "original-id", RequestUntil: time.Now().Add(time.Second), WriteUntil: time.Now().Add(2 * time.Second), Lifecycle: &audit.OperationLifecycleBinding{}}
	receipt := &audit.Receipt{Key: "original-receipt"}
	out := audit.OperationOutcome{Intent: in, Effect: "unknown"}
	snap := audit.OperationSnapshot{Intent: in, IntentStarted: true, IntentReceipt: receipt, Outcome: &out}
	port := &nativeHTTPTestJournal{ctx: ctx, in: in, out: out, snapshot: snap, same: true, err: err}
	trace := &nativeHTTPTrace{start: time.Now()}
	wrapper := nativeHTTPJournal{port, trace}
	got, e := wrapper.ReserveOperation(ctx, in)
	if e != err || !reflect.DeepEqual(got, snap) || got.IntentReceipt != receipt || got.Outcome != &out {
		t.Fatal("audit snapshot changed")
	}
	fresh, e := wrapper.ClaimOperationWrite(ctx, in, "sensitive-unknown-phase")
	if fresh || e != err {
		t.Fatal("unknown claim result became fresh work")
	}
	got, e = wrapper.FreezeOperationOutcome(ctx, out)
	if e != err || !reflect.DeepEqual(got, snap) || !port.same || port.calls != [3]int{1, 1, 1} {
		t.Fatal("trace changed original audit call/result")
	}
	if trace.used != 3 || trace.events[1].phase != "http.audit-invalid-phase" || trace.events[1].contextResult != "context-canceled" {
		t.Fatal("unexpected phase was logged or context changed")
	}
}
func TestNativeHTTPTraceBoundedIndependentBuffer(t *testing.T) {
	trace := &nativeHTTPTrace{start: time.Now()}
	var wg sync.WaitGroup
	for n := 0; n < nativeHTTPTraceLimit; n++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			trace.begin(context.Background(), "http.authority-current", time.Time{})(nil, time.Time{})
		}()
	}
	wg.Wait()
	trace.begin(context.Background(), "http.audit-reserve", time.Time{})(artifact.ErrDownloadUnavailable, time.Time{})
	trace.begin(context.Background(), "http.audit-freeze", time.Time{})(context.Canceled, time.Time{})
	if trace.used != nativeHTTPTraceLimit || trace.omitted != 2 || trace.firstError == nil || trace.firstError.phase != "http.audit-reserve" || trace.firstError.result != "download-unavailable" {
		t.Fatal("bounded HTTP trace lost first returned failure after overflow")
	}
	for _, e := range trace.events {
		if e.ended.IsZero() {
			t.Fatal("trace call not joined")
		}
	}
}
