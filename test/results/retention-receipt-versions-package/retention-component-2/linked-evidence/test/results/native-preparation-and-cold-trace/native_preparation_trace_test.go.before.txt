// This fixture-only trace forwards the original calls without new contexts,
// checks, retries, goroutines or authority. It retains only fixed phase labels,
// timing and closed result classifications, never input/result bodies or errors.
// Synchronous test output is deferred until the owning fixture has joined.
const nativePreparationTraceLimit = 512

type nativePreparationTraceEvent struct {
	phase                    string
	started, ended, deadline time.Time
	inputUntil, resultUntil  time.Time
	result, contextResult    string
}

type nativePreparationTrace struct {
	mu         sync.Mutex
	paused     bool
	start      time.Time
	events     [nativePreparationTraceLimit]nativePreparationTraceEvent
	used       int
	omitted    uint64
	firstError *nativePreparationTraceEvent
}

func nativePreparationResult(err error) string {
	switch {
	case err == nil:
		return "ok"
	case errors.Is(err, context.DeadlineExceeded):
		return "context-deadline"
	case errors.Is(err, context.Canceled):
		return "context-canceled"
	case errors.Is(err, artifact.ErrDownloadExpired):
		return "download-expired"
	case errors.Is(err, artifact.ErrDownloadUnavailable):
		return "download-unavailable"
	case errors.Is(err, artifact.ErrDownloadConflict):
		return "download-conflict"
	case errors.Is(err, artifact.ErrDownloadCapacity):
		return "download-capacity"
	case errors.Is(err, artifact.ErrInvalid):
		return "invalid"
	case errors.Is(err, artifact.ErrIntegrity):
		return "integrity"
	case errors.Is(err, identity.ErrDenied):
		return "identity-denied"
	case errors.Is(err, identity.ErrUnavailable):
		return "identity-unavailable"
	default:
		return "other-error"
	}
}

func (s *nativePreparationTrace) startCapture() {
	s.mu.Lock()
	s.start = time.Now()
	s.paused = false
	s.mu.Unlock()
}
func (s *nativePreparationTrace) stopCapture() {
	s.mu.Lock()
	s.paused = true
	s.mu.Unlock()
}
func (s *nativePreparationTrace) begin(ctx context.Context, phase string, inputUntil time.Time) func(error, time.Time) {
	s.mu.Lock()
	paused := s.paused
	s.mu.Unlock()
	if paused {
		return func(error, time.Time) {}
	}

	e := nativePreparationTraceEvent{phase: phase, started: time.Now(), inputUntil: inputUntil}
	if ctx != nil {
		e.deadline, _ = ctx.Deadline()
	}
	s.mu.Lock()
	index := -1
	if s.used < len(s.events) {
		index = s.used
		s.used++
		s.events[index] = e
	} else {
		s.omitted++
	}
	s.mu.Unlock()
	return func(err error, until time.Time) {
		e.ended, e.resultUntil, e.result = time.Now(), until, nativePreparationResult(err)
		e.contextResult = "none"
		if ctx != nil {
			e.contextResult = nativePreparationResult(ctx.Err())
		}
		s.mu.Lock()
		defer s.mu.Unlock()
		if index >= 0 {
			s.events[index] = e
		}
		if err != nil && s.firstError == nil {
			copy := e
			s.firstError = &copy
		}
	}
}

func nativePreparationRemaining(until, at time.Time) time.Duration {
	if until.IsZero() {
		return 0
	}
	return until.Sub(at)
}

func (s *nativePreparationTrace) log(t *testing.T) {
	t.Helper()
	s.mu.Lock()
	events := append([]nativePreparationTraceEvent(nil), s.events[:s.used]...)
	omitted, first := s.omitted, s.firstError
	s.mu.Unlock()
	for i, e := range events {
		if e.ended.IsZero() {
			t.Logf("native preparation trace phase=%s seq=%d start=%s result=not-joined", e.phase, i, e.started.Sub(s.start))
			continue
		}
		t.Logf("native preparation trace phase=%s seq=%d start=%s elapsed=%s deadline=%s remaining-start=%s remaining-end=%s input-until=%s input-remaining-end=%s result-until=%s result-remaining-end=%s result=%s context=%s",
			e.phase, i, e.started.Sub(s.start), e.ended.Sub(e.started), e.deadline.UTC().Format(time.RFC3339Nano), nativePreparationRemaining(e.deadline, e.started), nativePreparationRemaining(e.deadline, e.ended), e.inputUntil.UTC().Format(time.RFC3339Nano), nativePreparationRemaining(e.inputUntil, e.ended), e.resultUntil.UTC().Format(time.RFC3339Nano), nativePreparationRemaining(e.resultUntil, e.ended), e.result, e.contextResult)
	}
	if first != nil {
		t.Logf("native preparation trace first-error phase=%s elapsed=%s result=%s context=%s", first.phase, first.ended.Sub(first.started), first.result, first.contextResult)
	}
	t.Logf("native preparation trace retained=%d omitted=%d", len(events), omitted)
}

func traceNativePreparation(t *testing.T, c evidence.ContextNativeRuntimeConfig) (evidence.ContextNativeRuntimeConfig, *nativePreparationTrace) {
	t.Helper()
	s := &nativePreparationTrace{start: time.Now(), paused: true}
	t.Cleanup(func() { s.log(t) })
	// Copy the actual composed authority; wrappers never substitute fixture facts.
	a, ok := c.Authority.(authapp.ContextDownloadAuthority)
	if !ok {
		t.Fatal("native preparation trace requires the actual current authority")
	}
	a.Permissions = nativeTracePermissions{a.Permissions, s}
	a.Catalog = nativeTraceCatalog{a.Catalog, s}
	a.Retention = nativeTraceRetention{a.Retention, s}
	a.Environment = nativeTraceEnvironment{a.Environment, s}
	c.Authority = nativeTraceAuthority{a, s}
	c.Workers = nativeTraceAttachments{c.Workers, s}
	c.Sources = nativeTraceSources{c.Sources, s}
	c.Preparation = nativeTracePreparation{c.Preparation, s}
	c.Audit = nativeTraceAudit{c.Audit, s}
	c.Acknowledgements = nativeTraceAcknowledgements{c.Acknowledgements, s}
	d := *c.Dispatcher
	d.Ledger = nativeTraceWorkerLedger{d.Ledger, s}
	d.Jobs = nativeTraceJobs{d.Jobs, s}
	c.Dispatcher = &d
	r := *c.Requests
	r.Store = nativeTraceWorkerOwners{r.Store, s}
	c.Requests = &r
	c.Native.Fences = nativeTraceFences{c.Native.Fences, s}
	owners, ok := c.Native.Owners.(nativeTraceFirstOperations)
	if !ok {
		t.Fatal("native preparation trace requires the original first-context owner facet")
	}
	c.Native.Owners = nativeTraceOperations{owners, s}
	c.Native.Origins.Operations = nativeTraceOperationReader{c.Native.Origins.Operations, s}
	c.Native.PreparationKeys = nativeTracePreparationKeys{c.Native.PreparationKeys, s}
	c.Native.Registrar = nativeTraceRegistrar{c.Native.Registrar, s}
	return c, s
}

func (s nativeTraceRegistrar) RegisterNativeExchange(ctx context.Context, in evidence.NativeExchangeRegistration, before func(context.Context, evidence.NativeOrigin) error, check func(context.Context, evidence.NativeOrigin) (value.NativeLease, error)) (evidence.NativePendingExchange, error) {
	done := s.trace.begin(ctx, "native.register", in.Operation.Until)
	v, e := s.NativeExchangeRegistrar.RegisterNativeExchange(ctx, in, func(call context.Context, origin evidence.NativeOrigin) error {
		finished := s.trace.begin(call, "native.before-command", origin.Lease.Until)
		err := before(call, origin)
		finished(err, time.Time{})
		return err
	}, func(call context.Context, origin evidence.NativeOrigin) (value.NativeLease, error) {
		finished := s.trace.begin(call, "native.current-frame", origin.Lease.Until)
		lease, err := check(call, origin)
		finished(err, lease.Until)
		return lease, err
	})
	done(e, time.Time{})
	if v != nil {
		v = nativeTracePending{v, s.trace}
	}
	return v, e
}

// These tests only verify that diagnostics do not alter calls or log raw errors.
// They do not replace the actual SQL/S3/native preparation qualification.
func TestNativePreparationTraceForwardsOriginalCall(t *testing.T) {
	ctx, cancel := context.WithDeadline(context.Background(), time.Now().Add(time.Second))
	defer cancel()
	err := errors.New("sensitive error text must never enter trace")
	until := time.Now().Add(500 * time.Millisecond)
	underlying := &nativeTraceTestAuthority{context: ctx, result: artifact.DownloadAuthority{ValidUntil: until}, err: err}
	trace := &nativePreparationTrace{start: time.Now()}
	wrapper := nativeTraceAuthority{underlying, trace}
	got, gotErr := wrapper.CheckDownloadCurrent(ctx, nil, artifact.DownloadScope{})
	if got != underlying.result || gotErr != err || underlying.calls != 1 || !underlying.sameContext {
		t.Fatal("trace changed original call, result, error or invocation count")
	}
	e := trace.events[0]
	deadline, _ := ctx.Deadline()
	if trace.used != 1 || e.deadline != deadline || e.resultUntil != until || e.result != "other-error" || e.contextResult != "ok" {
		t.Fatal("trace changed deadline or retained non-closed diagnostics")
	}
	cancel()
	_, gotErr = wrapper.CheckDownloadCurrent(ctx, nil, artifact.DownloadScope{})
	if gotErr != err || underlying.calls != 2 || trace.events[1].contextResult != "context-canceled" {
		t.Fatal("trace intercepted canceled call or changed original error")
	}
}

type nativeTraceTestAuthority struct {
	context     context.Context
	result      artifact.DownloadAuthority
	err         error
	calls       int
	sameContext bool
}

func (s *nativeTraceTestAuthority) CheckDownloadCurrent(ctx context.Context, _ identities.CurrentIdentity, _ artifact.DownloadScope) (artifact.DownloadAuthority, error) {
	s.calls++
	s.sameContext = ctx == s.context
	return s.result, s.err
}

func TestNativePreparationTraceRetainsBoundedEventsAndFirstError(t *testing.T) {
	trace := &nativePreparationTrace{start: time.Now()}
	for n := 0; n < nativePreparationTraceLimit+2; n++ {
		err := error(nil)
		if n == nativePreparationTraceLimit {
			err = artifact.ErrDownloadUnavailable
		}
		trace.begin(context.Background(), "authority.current", time.Time{})(err, time.Time{})
	}
	if trace.used != nativePreparationTraceLimit || trace.omitted != 2 || trace.firstError == nil || trace.firstError.result != "download-unavailable" {
		t.Fatal("trace must stay bounded and preserve first error after overflow")
	}
}

func TestNativePreparationTracePausedStillForwards(t *testing.T) {
	ctx := context.Background()
	expected := errors.New("private original error")
	underlying := &nativeTraceTestAuthority{context: ctx, err: expected}
	trace := &nativePreparationTrace{paused: true}
	wrapper := nativeTraceAuthority{underlying, trace}
	if _, e := wrapper.CheckDownloadCurrent(ctx, nil, artifact.DownloadScope{}); e != expected || trace.used != 0 || underlying.calls != 1 || !underlying.sameContext {
		t.Fatal("paused trace changed original call")
	}
	trace.startCapture()
	if _, e := wrapper.CheckDownloadCurrent(ctx, nil, artifact.DownloadScope{}); e != expected || trace.used != 1 || underlying.calls != 2 {
		t.Fatal("active trace changed original call")
	}
	trace.stopCapture()
	if _, e := wrapper.CheckDownloadCurrent(ctx, nil, artifact.DownloadScope{}); e != expected || trace.used != 1 || underlying.calls != 3 {
		t.Fatal("stopped trace changed original call")
	}
}
