import hashlib,json,os,subprocess,time
from pathlib import Path
root=Path('/home/phil/Projekte/asservat')
stage=Path(__file__).resolve().parent
env=dict(os.environ,GOMAXPROCS='2',GOFLAGS='-p=2',TMPDIR='/home/phil/.cache/asservat-build/tmp')
def hashes():
    paths=[root/'go.mod',root/'go.sum']
    for folder,glob in [('cmd','*.go'),('internal','*.go'),('internal','*.json'),('test/integration','*.go'),('migrations','*.sql')]:
        paths+=list((root/folder).rglob(glob))
    return {str(p.relative_to(root)):hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted(paths)}
original=hashes()
(stage/'component-sources.json').write_text(json.dumps(original,indent=2)+'\n')
results=[]
for name,command in [('compile',['go','test','-overlay='+str(stage/'overlay.json'),'-run','^$','./test/integration']),('vet',['go','vet','-overlay='+str(stage/'overlay.json'),'./test/integration'])]:
    began=time.monotonic()
    with (stage/(name+'.log')).open('w') as log:
        result=subprocess.run(command,cwd=root,env=env,stdout=log,stderr=subprocess.STDOUT)
    current=hashes()
    changed=[p for p in sorted(set(original)|set(current)) if original.get(p)!=current.get(p)]
    results.append(dict(name=name,command=command,exit=result.returncode,seconds=round(time.monotonic()-began,3),changedSources=changed))
    (stage/'component-results.json').write_text(json.dumps(results,indent=2)+'\n')
    print(json.dumps(results[-1]),flush=True)
    if result.returncode or changed: raise SystemExit(result.returncode or 1)
