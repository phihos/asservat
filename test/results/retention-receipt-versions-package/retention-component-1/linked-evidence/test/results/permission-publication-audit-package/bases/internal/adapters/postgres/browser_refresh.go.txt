// BrowserRefreshOwners composes already qualified browser, current C14 rights,
// retention/recovery transactions and actual process-owned serving identity.
// It introduces no identity evaluator or capacity configuration. Admission,
// maintenance and read-only pause facets use purpose-scoped SQL credentials.
type BrowserRefreshDatabases struct{ Admission, Maintenance, Reader *BrowserSessions }

type BrowserRefreshOwners struct {
	databases      BrowserRefreshDatabases
	sessions       *BrowserSessions
	downloads      *Downloads
	authority      evidence.CurrentDownloadAuthority
	serving        fleetapp.OwnerServingIdentity
	ownerAuthority *fleetapp.OwnerAuthenticator
	binding        value.RefreshProcessBinding
	now            func() time.Time
}

func NewBrowserRefreshOwners(databases BrowserRefreshDatabases, downloads *Downloads, authority evidence.CurrentDownloadAuthority, serving fleetapp.OwnerServingIdentity, ownerAuthority *fleetapp.OwnerAuthenticator, binding value.RefreshProcessBinding, now func() time.Time) (*BrowserRefreshOwners, error) {
	sessions := databases.Admission
	if sessions == nil || databases.Maintenance == nil || databases.Reader == nil || downloads == nil || downloads.retention == nil || downloads.recovery == nil || authority == nil || serving == nil || ownerAuthority == nil || !binding.Valid() || binding.RootID != sessions.root || binding.Epoch != sessions.epoch || downloads.recovery.installation != binding.InstallationID || now == nil {
		return nil, value.ErrInvalid
	}
	for _, db := range []*BrowserSessions{databases.Maintenance, databases.Reader} {
		if db.binding != sessions.binding {
			return nil, value.ErrInvalid
		}
	}
	return &BrowserRefreshOwners{databases, sessions, downloads, authority, serving, ownerAuthority, binding, now}, nil
}

func refreshError(err error) error {
	if err == nil {
		return nil
	}
	for _, v := range []error{value.ErrDenied, value.ErrSessionBusy, value.ErrSessionCapacity} {
		if errors.Is(err, v) {
			return v
		}
	}
	return value.ErrUnavailable
}
func (s *BrowserRefreshOwners) process(ctx context.Context) (time.Time, error) {
	until, err := s.ownerAuthority.CheckOwnerServing(ctx, s.serving)
	if err != nil {
		return time.Time{}, value.ErrUnavailable
	}
	cert, err := s.serving.ServingCertificate()
	b := s.binding
	if err != nil || !cert.Valid(s.now()) || cert.Binding.InstallationID != b.InstallationID || cert.Binding.ClusterID != b.ClusterID || cert.Binding.PodUID != b.PodUID || cert.Binding.ProcessEpoch != b.ProcessEpoch || ctx.Err() != nil || !s.now().Before(until) {
		return time.Time{}, value.ErrDenied
	}
	return value.Earlier(until, cert.NotAfter), nil
}
func refreshRef(r custodydb.BrowserRefreshOwner) value.RefreshOwnerRef {
	v := value.RefreshOwnerRef{RootID: uid(r.RootID), Epoch: uid(r.Epoch), OwnerID: uid(r.OwnerID), InstallationID: uid(r.InstallationID), ClusterID: uid(r.ClusterID), PodUID: uid(r.PodUid), ProcessEpoch: uid(r.ProcessEpoch), Generation: r.Generation}
	copy(v.SessionID[:], r.SessionID)
	return v
}
func refreshOwnerView(r custodydb.BrowserRefreshOwner, observed, until, currentEnd time.Time) value.RefreshOwnerObservation {
	// Nested process/SQL observations may start later. Their deadlines must not
	// lengthen the original caller's five-second observation interval.
	until = value.Earlier(until, observed.Add(5*time.Second))
	return value.RefreshOwnerObservation{Owner: refreshRef(r), SessionEnd: value.Earlier(r.SessionExpiresAt.Time, currentEnd), LeaseUntil: value.Earlier(r.LeaseUntil.Time, currentEnd), ObservedAt: observed, ValidUntil: value.Earlier(until, value.Earlier(r.LeaseUntil.Time, currentEnd))}
}
func refreshOwnerSession(r custodydb.BrowserRefreshOwner, s custodydb.BrowserSession, now time.Time) bool {
	pin := browserRecord(s)
	provider := sha256.Sum256([]byte(pin.ProviderSession))
	return pin.Valid(now) && r.Epoch == s.Epoch && bytes.Equal(r.SessionID, s.ID) && bytes.Equal(r.ActorKey, digest(pin.Principal)) && bytes.Equal(r.ProviderSessionKey, provider[:]) && r.SessionCreatedAt.Time.Equal(s.CreatedAt.Time) && r.AuthenticatedAt.Time.Equal(s.AuthenticatedAt.Time) && !s.ExpiresAt.Time.After(r.SessionExpiresAt.Time)
}
func (s *BrowserRefreshOwners) root(ctx context.Context, q *custodydb.Queries) (custodydb.BrowserSessionRoot, error) {
	r, e := q.LockBrowserRoot(ctx, id(s.sessions.root))
	if e != nil {
		return r, e
	}
	if uid(r.Epoch) != s.sessions.epoch || r.Issuer.String != s.sessions.binding.Issuer || r.ClientID.String != s.sessions.binding.ClientID || r.Origin.String != s.sessions.binding.Origin {
		return r, value.ErrDenied
	}
	return r, nil
}
func refreshCurrent(ctx context.Context, q *custodydb.Queries, root custodydb.BrowserSessionRoot, session custodydb.BrowserSession, until time.Time) (time.Time, error) {
	n, e := q.CurrentTime(ctx)
	if e != nil {
		return time.Time{}, e
	}
	if ctx.Err() != nil || !n.Time.Before(until) || !browserSettingsCurrent(root, n.Time) || !browserRecord(session).Valid(n.Time) || root.Epoch != session.Epoch {
		return time.Time{}, value.ErrDenied
	}
	return n.Time, nil
}

// effect is the exact current ordinary download gate, before and after locks.
// Human identity and network/source resolution run before acquiring SQL locks.
func (s *BrowserRefreshOwners) effect(ctx context.Context, database *BrowserSessions, current identities.CurrentIdentity, pin value.RefreshSessionPin, downloadID string, fn func(*custodydb.Queries, custodydb.BrowserSessionRoot, custodydb.BrowserSession, artifact.DownloadSession, artifact.DownloadAuthority, time.Time, time.Time) error) error {
	if ctx == nil || current == nil || ctx.Err() != nil {
		return value.ErrDenied
	}
	start := s.now()
	ctx, cancel := context.WithDeadline(ctx, start.Add(5*time.Second))
	defer cancel()
	i, e := current.ResolveIdentity(ctx)
	if e != nil || !i.Valid(s.now()) || i.Principal != pin.Principal {
		return value.ErrDenied
	}
	d, e := s.downloads.LookupDownload(ctx, downloadID, i.Principal)
	if e != nil || d.State.Terminal() {
		return value.ErrDenied
	}
	a, e := s.authority.CheckDownloadCurrent(ctx, current, d.Scope)
	if e != nil || !a.Valid(s.now()) || a.Scope != d.Scope || !a.Environment.Valid(s.now()) {
		return value.ErrDenied
	}
	processUntil, e := s.process(ctx)
	if e != nil {
		return e
	}
	topology, e := s.downloads.retention.observe(ctx, a.Scope.ClusterID)
	if e != nil {
		return e
	}
	final, e := current.ResolveIdentity(ctx)
	if e != nil || final.Principal != i.Principal || !final.Valid(s.now()) {
		return value.ErrDenied
	}
	until := value.Earlier(start.Add(5*time.Second), value.Earlier(processUntil, value.Earlier(i.ValidUntil, value.Earlier(final.ValidUntil, a.ValidUntil))))
	return database.ledger.transaction(ctx, func(q *custodydb.Queries) error {
		recovery, e := s.downloads.recovery.lock(ctx, q, s.downloads.root, a.Environment.Recovery())
		if e != nil {
			return e
		}
		root, e := s.root(ctx, q)
		if e != nil {
			return e
		}
		browser, e := q.LockBrowserSession(ctx, custodydb.LockBrowserSessionParams{RootID: id(s.sessions.root), Epoch: id(s.sessions.epoch), ID: pin.ID[:]})
		if e != nil {
			return e
		}
		record := browserRecord(browser)
		record.Observation = sessionObservation(root, record.ExpiresAt)
		if !pin.Matches(record, s.now()) {
			return value.ErrDenied
		}
		if _, e = q.LockBrowserRefreshOwner(ctx, custodydb.LockBrowserRefreshOwnerParams{RootID: id(s.sessions.root), SessionID: pin.ID[:]}); e != nil && !errors.Is(e, pgx.ErrNoRows) {
			return e
		}
		permission, e := q.LockPermissionStream(ctx, id(s.downloads.root))
		if e != nil {
			return e
		}
		row, e := q.LockDownloadSession(ctx, custodydb.LockDownloadSessionParams{ID: id(downloadID), RootID: id(s.downloads.root)})
		if e != nil {
			return e
		}
		stored, e := decodeDownload(row)
		if e != nil || stored.Scope != a.Scope || stored.State.Terminal() {
			return value.ErrDenied
		}
		n, e := refreshCurrent(ctx, q, root, browser, until)
		if e != nil || !n.Before(stored.Deadline) {
			return value.ErrDenied
		}
		if e = s.downloads.downloadFence(ctx, q, permission, a, topology, recovery); e != nil {
			return e
		}
		if e = fn(q, root, browser, stored, a, n, until); e != nil {
			return e
		}
		if _, e = refreshCurrent(ctx, q, root, browser, until); e != nil {
			return e
		}
		return s.downloads.downloadFence(ctx, q, permission, a, topology, recovery)
	})
}

func (s *BrowserRefreshOwners) AdmitBrowserRefresh(ctx context.Context, current identities.CurrentIdentity, pin value.RefreshSessionPin, proposed value.RefreshOwnerRef, downloadID string) (value.RefreshOwnerObservation, error) {
	var out value.RefreshOwnerObservation
	if !s.binding.Owns(proposed) || pin.ID != proposed.SessionID {
		return out, value.ErrDenied
	}
	started := s.now()
	e := s.effect(ctx, s.databases.Admission, current, pin, downloadID, func(q *custodydb.Queries, root custodydb.BrowserSessionRoot, browser custodydb.BrowserSession, d artifact.DownloadSession, a artifact.DownloadAuthority, n, until time.Time) error {
		old, e := q.LockBrowserRefreshOwner(ctx, custodydb.LockBrowserRefreshOwnerParams{RootID: id(proposed.RootID), SessionID: pin.ID[:]})
		if e != nil && !errors.Is(e, pgx.ErrNoRows) {
			return e
		}
		fresh := errors.Is(e, pgx.ErrNoRows)
		if !fresh && !refreshOwnerSession(old, browser, n) {
			return value.ErrDenied
		}
		lease := value.Earlier(started.Add(value.RefreshOwnerLease), browser.ExpiresAt.Time)
		if fresh {
			provider := sha256.Sum256([]byte(pin.ProviderSession))
			old, e = q.InsertBrowserRefreshOwner(ctx, custodydb.InsertBrowserRefreshOwnerParams{RootID: id(proposed.RootID), SessionID: pin.ID[:], Epoch: id(proposed.Epoch), OwnerID: id(proposed.OwnerID), InstallationID: id(proposed.InstallationID), ClusterID: id(proposed.ClusterID), PodUid: id(proposed.PodUID), ProcessEpoch: id(proposed.ProcessEpoch), ActorKey: digest(pin.Principal), ProviderSessionKey: provider[:], SessionCreatedAt: browser.CreatedAt, AuthenticatedAt: browser.AuthenticatedAt, SessionExpiresAt: browser.ExpiresAt, StartedAt: ts(n), LeaseUntil: ts(lease)})
		} else if old.JoinedAt.Valid {
			old, e = q.ReassignJoinedBrowserRefreshOwner(ctx, custodydb.ReassignJoinedBrowserRefreshOwnerParams{RootID: old.RootID, SessionID: old.SessionID, OwnerID: old.OwnerID, OwnerID_2: id(proposed.OwnerID), InstallationID: id(proposed.InstallationID), ClusterID: id(proposed.ClusterID), PodUid: id(proposed.PodUID), ProcessEpoch: id(proposed.ProcessEpoch), StartedAt: ts(n), LeaseUntil: ts(lease), SessionExpiresAt: browser.ExpiresAt})
		} else if !s.binding.Owns(refreshRef(old)) || !n.Before(old.LeaseUntil.Time) {
			return value.ErrSessionBusy
		}
		if e != nil {
			return e
		}
		consumer, e := q.LockBrowserRefreshConsumer(ctx, id(downloadID))
		if errors.Is(e, pgx.ErrNoRows) {
			hard := value.Earlier(browser.ExpiresAt.Time, value.Earlier(d.Deadline, a.AccessUntil))
			_, e = q.InsertBrowserRefreshConsumer(ctx, custodydb.InsertBrowserRefreshConsumerParams{DownloadID: id(downloadID), DownloadRootID: id(d.Scope.RootID), FirstAdmittedAt: ts(n), GraceUntil: ts(value.Earlier(n.Add(value.RefreshAttachmentGrace), hard)), HardUntil: ts(hard), ActorKey: digest(pin.Principal), OwnerID: old.OwnerID, OwnerGeneration: pgtype.Int8{Int64: old.Generation, Valid: true}})
		} else if e == nil {
			if consumer.DownloadRootID != id(d.Scope.RootID) || !bytes.Equal(consumer.ActorKey, digest(pin.Principal)) || !n.Before(consumer.HardUntil.Time) {
				return value.ErrDenied
			}
			if !consumer.OwnerID.Valid {
				_, e = q.AttachBrowserRefreshConsumer(ctx, custodydb.AttachBrowserRefreshConsumerParams{DownloadID: consumer.DownloadID, OwnerID: old.OwnerID, OwnerGeneration: pgtype.Int8{Int64: old.Generation, Valid: true}})
			} else if consumer.OwnerID != old.OwnerID || consumer.OwnerGeneration.Int64 != old.Generation {
				return value.ErrSessionBusy
			}
		}
		if e != nil {
			return e
		}
		out = refreshOwnerView(old, started, until, browser.ExpiresAt.Time)
		return nil
	})
	if e != nil || ctx.Err() != nil || !out.Valid(s.now()) {
		return value.RefreshOwnerObservation{}, refreshError(eOrDenied(e))
	}
	return out, nil
}
func eOrDenied(e error) error {
	if e == nil {
		return value.ErrDenied
	}
	return e
}

func (s *BrowserRefreshOwners) maintenance(ctx context.Context, ref value.RefreshOwnerRef, current bool, fn func(*custodydb.Queries, custodydb.BrowserRefreshOwner, custodydb.BrowserSession, time.Time, time.Time) error) error {
	if ctx == nil || ctx.Err() != nil || !s.binding.Owns(ref) {
		return value.ErrDenied
	}
	start := s.now()
	until := start.Add(5 * time.Second)
	ctx, cancel := context.WithDeadline(ctx, until)
	defer cancel()
	if current {
		u, e := s.process(ctx)
		if e != nil {
			return e
		}
		until = value.Earlier(until, u)
	}
	return s.databases.Maintenance.ledger.transaction(ctx, func(q *custodydb.Queries) error {
		var trust custodydb.FleetCurrentTrust
		var err error
		if current {
			trust, err = q.ShareFleetTrust(ctx, id(s.binding.InstallationID))
			if err != nil {
				return err
			}
		}
		var root custodydb.BrowserSessionRoot
		if current {
			root, err = s.root(ctx, q)
		} else {
			root, err = q.LockBrowserRoot(ctx, id(ref.RootID))
		}
		if err != nil {
			return err
		}
		var browser custodydb.BrowserSession
		if current {
			browser, err = q.LockBrowserSession(ctx, custodydb.LockBrowserSessionParams{RootID: id(ref.RootID), Epoch: id(ref.Epoch), ID: ref.SessionID[:]})
			if err != nil {
				return err
			}
			// Return observations with the same original settings/session bounds
			// checked inside this transaction, including after lock waits.
			until = value.Earlier(until, value.Earlier(root.FreshUntil.Time, browser.ExpiresAt.Time))
		}
		owner, err := q.LockBrowserRefreshOwner(ctx, custodydb.LockBrowserRefreshOwnerParams{RootID: id(ref.RootID), SessionID: ref.SessionID[:]})
		if !current && errors.Is(err, pgx.ErrNoRows) {
			return nil
		}
		if err != nil {
			return err
		}
		if refreshRef(owner) != ref {
			return value.ErrDenied
		}
		check := func() (time.Time, error) {
			n, e := q.CurrentTime(ctx)
			if e != nil {
				return time.Time{}, e
			}
			if ctx.Err() != nil || !n.Time.Before(until) {
				return time.Time{}, value.ErrDenied
			}
			if current {
				if owner.JoinedAt.Valid || !n.Time.Before(owner.LeaseUntil.Time) || !refreshOwnerSession(owner, browser, n.Time) {
					return time.Time{}, value.ErrDenied
				}
				if _, e = refreshCurrent(ctx, q, root, browser, until); e != nil {
					return time.Time{}, e
				}
				recovery, e := recoveryState(trust, start, n.Time)
				if e != nil {
					return time.Time{}, e
				}
				until = value.Earlier(until, recovery.FreshUntil)
			}
			return n.Time, nil
		}
		n, err := check()
		if err != nil {
			return err
		}
		if err = fn(q, owner, browser, n, until); err != nil {
			return err
		}
		_, err = check()
		return err
	})
}

func (s *BrowserRefreshOwners) demand(ctx context.Context, q *custodydb.Queries, owner custodydb.BrowserRefreshOwner, browser custodydb.BrowserSession, observed, until time.Time) (value.RefreshDemand, error) {
	consumers, e := q.ReadBrowserRefreshConsumers(ctx, custodydb.ReadBrowserRefreshConsumersParams{OwnerID: owner.OwnerID, OwnerGeneration: pgtype.Int8{Int64: owner.Generation, Valid: true}})
	if e != nil {
		return value.RefreshDemand{}, e
	}
	if len(consumers) > value.MaxRefreshConsumers {
		return value.RefreshDemand{}, value.ErrDenied
	}
	var chosen value.RefreshDemand
	for _, consumer := range consumers {
		row, e := q.LockDownloadSession(ctx, custodydb.LockDownloadSessionParams{ID: consumer.DownloadID, RootID: consumer.DownloadRootID})
		if e != nil {
			return value.RefreshDemand{}, e
		}
		d, e := decodeDownload(row)
		if e != nil || consumer.DownloadRootID != id(s.downloads.root) || !bytes.Equal(consumer.ActorKey, owner.ActorKey) || !bytes.Equal(row.ActorKey, owner.ActorKey) {
			return value.RefreshDemand{}, value.ErrDenied
		}
		n, e := q.CurrentTime(ctx)
		if e != nil {
			return value.RefreshDemand{}, e
		}
		hard := value.Earlier(consumer.HardUntil.Time, value.Earlier(browser.ExpiresAt.Time, d.Deadline))
		if d.State.Terminal() || !n.Time.Before(hard) {
			if !owner.ClaimDownloadID.Valid || owner.ClaimDownloadID != consumer.DownloadID || owner.ClaimJoinedAt.Valid {
				if _, e = q.DetachBrowserRefreshConsumer(ctx, custodydb.DetachBrowserRefreshConsumerParams{DownloadID: consumer.DownloadID, OwnerID: owner.OwnerID, OwnerGeneration: pgtype.Int8{Int64: owner.Generation, Valid: true}}); e != nil {
					return value.RefreshDemand{}, e
				}
			}
			continue
		}
		// Effective publication retention remains a scheduling cutoff; fresh full
		// retention/rights/recovery authority is independently required at claim.
		published, e := q.ReadPublishedContext(ctx, custodydb.ReadPublishedContextParams{ArtifactID: id(d.Scope.ArtifactID), RootID: consumer.DownloadRootID})
		if e != nil {
			return value.RefreshDemand{}, e
		}
		hard = value.Earlier(hard, published.EffectiveExpiresAt.Time)
		if !n.Time.Before(hard) {
			continue
		}
		v := value.RefreshDemand{Owner: refreshOwnerView(owner, observed, until, browser.ExpiresAt.Time), DownloadID: d.ID, HardUntil: hard, GraceUntil: value.Earlier(consumer.GraceUntil.Time, hard)}
		response, e := q.ReadCurrentBrowserRefreshResponse(ctx, custodydb.ReadCurrentBrowserRefreshResponseParams{SessionID: consumer.DownloadID, RootID: consumer.DownloadRootID})
		if e == nil {
			v.ResponseID = uid(response.ID)
			v.ResponseOwnerID = uid(response.OwnerID)
			v.ResponseGeneration = response.Generation
			v.ResponseLeaseUntil = value.Earlier(response.LeaseUntil.Time, observed.Add(5*time.Second))
		} else if !errors.Is(e, pgx.ErrNoRows) {
			return value.RefreshDemand{}, e
		}
		if chosen.DownloadID == "" && v.Valid(n.Time) {
			chosen = v
		}
	}
	if chosen.Valid(s.now()) {
		return chosen, nil
	}
	return value.RefreshDemand{}, value.ErrDenied
}

func (s *BrowserRefreshOwners) ObserveBrowserRefresh(ctx context.Context, ref value.RefreshOwnerRef) (value.RefreshDemand, error) {
	var out value.RefreshDemand
	started := s.now()
	e := s.maintenance(ctx, ref, true, func(q *custodydb.Queries, owner custodydb.BrowserRefreshOwner, browser custodydb.BrowserSession, n, until time.Time) error {
		d, e := s.demand(ctx, q, owner, browser, started, until)
		if e != nil {
			return e
		}
		lease := value.Earlier(started.Add(value.RefreshOwnerLease), value.Earlier(browser.ExpiresAt.Time, owner.SessionExpiresAt.Time))
		owner, e = q.RenewBrowserRefreshOwner(ctx, custodydb.RenewBrowserRefreshOwnerParams{RootID: owner.RootID, SessionID: owner.SessionID, OwnerID: owner.OwnerID, Generation: owner.Generation, LeaseUntil: ts(lease)})
		if e != nil {
			return e
		}
		d.Owner = refreshOwnerView(owner, started, until, browser.ExpiresAt.Time)
		out = d
		return nil
	})
	if e != nil || ctx.Err() != nil || !out.Valid(s.now()) {
		return value.RefreshDemand{}, refreshError(eOrDenied(e))
	}
	return out, nil
}

func (s *BrowserRefreshOwners) ClaimBrowserOwnedRefresh(ctx context.Context, current identities.CurrentIdentity, demand value.RefreshDemand, version int64, claim [32]byte) error {
	if !demand.Valid(s.now()) || !s.binding.Owns(demand.Owner.Owner) || version < 1 || claim == ([32]byte{}) {
		return value.ErrDenied
	}
	ref := demand.Owner.Owner
	browser, e := s.databases.Maintenance.ReadSession(ctx, ref.SessionID)
	if e != nil {
		return e
	}
	pin := value.RefreshSessionPin{ID: browser.ID, Principal: browser.Principal, ProviderSession: browser.ProviderSession, CreatedAt: browser.CreatedAt, AuthenticatedAt: browser.AuthenticatedAt, ExpiresAt: browser.ExpiresAt}
	e = s.effect(ctx, s.databases.Maintenance, current, pin, demand.DownloadID, func(q *custodydb.Queries, root custodydb.BrowserSessionRoot, browser custodydb.BrowserSession, d artifact.DownloadSession, a artifact.DownloadAuthority, n, until time.Time) error {
		if !demand.Valid(n) || browser.Version != version || len(browser.RefreshClaim) > 0 {
			return value.ErrSessionBusy
		}
		owner, e := q.LockBrowserRefreshOwner(ctx, custodydb.LockBrowserRefreshOwnerParams{RootID: id(ref.RootID), SessionID: ref.SessionID[:]})
		if e != nil {
			return e
		}
		if refreshRef(owner) != ref || owner.JoinedAt.Valid || !refreshOwnerSession(owner, browser, n) || !n.Before(owner.LeaseUntil.Time) || len(owner.Claim) > 0 && !owner.ClaimJoinedAt.Valid {
			return value.ErrDenied
		}
		selected, e := s.demand(ctx, q, owner, browser, n, until)
		if e != nil {
			return e
		}
		if selected.DownloadID != demand.DownloadID || selected.ResponseID != demand.ResponseID || selected.ResponseOwnerID != demand.ResponseOwnerID || selected.ResponseGeneration != demand.ResponseGeneration {
			return value.ErrDenied
		}
		// Once admitted the original claim may bridge a short no-output pause; it
		// cannot extend browser/download/access/owner lifetime or be spent twice.
		claimUntil := value.Earlier(n.Add(value.RefreshClaimWindow), value.Earlier(selected.HardUntil, value.Earlier(owner.LeaseUntil.Time, a.AccessUntil)))
		if !n.Before(claimUntil) {
			return value.ErrDenied
		}
		if e = q.ClaimBrowserRefresh(ctx, custodydb.ClaimBrowserRefreshParams{RootID: owner.RootID, Epoch: owner.Epoch, ID: owner.SessionID, RefreshClaim: claim[:], ClaimUntil: ts(claimUntil)}); e != nil {
			return e
		}
		var response, responseOwner pgtype.UUID
		var generation pgtype.Int8
		if selected.ResponseID != "" {
			response = id(selected.ResponseID)
			responseOwner = id(selected.ResponseOwnerID)
			generation = pgtype.Int8{Int64: selected.ResponseGeneration, Valid: true}
		}
		_, e = q.BindBrowserRefreshClaim(ctx, custodydb.BindBrowserRefreshClaimParams{RootID: owner.RootID, SessionID: owner.SessionID, OwnerID: owner.OwnerID, Generation: owner.Generation, Claim: claim[:], ClaimVersion: pgtype.Int8{Int64: version, Valid: true}, ClaimUntil: ts(claimUntil), ClaimDownloadID: id(selected.DownloadID), ClaimResponseID: response, ClaimResponseOwner: responseOwner, ClaimResponseGeneration: generation})
		return e
	})
	return refreshError(e)
}

func (s *BrowserRefreshOwners) JoinBrowserRefreshClaim(ctx context.Context, ref value.RefreshOwnerRef, version int64, claim [32]byte) error {
	if version < 1 || claim == ([32]byte{}) {
		return value.ErrInvalid
	}
	return refreshError(s.maintenance(ctx, ref, false, func(q *custodydb.Queries, owner custodydb.BrowserRefreshOwner, _ custodydb.BrowserSession, n, _ time.Time) error {
		if owner.ClaimVersion.Int64 != version || !bytes.Equal(owner.Claim, claim[:]) {
			// A definitely absent claim after the original call joined is safe to
			// acknowledge as no SQL spend. An unrelated/spent version never matches.
			if len(owner.Claim) == 0 {
				return nil
			}
			return value.ErrDenied
		}
		if owner.ClaimJoinedAt.Valid {
			return nil
		}
		_, e := q.JoinBrowserRefreshClaim(ctx, custodydb.JoinBrowserRefreshClaimParams{RootID: owner.RootID, SessionID: owner.SessionID, OwnerID: owner.OwnerID, Generation: owner.Generation, Claim: claim[:], ClaimVersion: pgtype.Int8{Int64: version, Valid: true}, ClaimJoinedAt: ts(n)})
		return e
	}))
}

// AbandonBrowserRefreshAdmission is historical no-start closure for the exact
// server proposal, called only after its synchronous admission callback joined.
// The shared browser-root lock serializes any previously sent INSERT/COMMIT.
// A missing row cannot race a later callback mutation: all mutations require
// this lock, and the returned callback sends no further statements.
func (s *BrowserRefreshOwners) AbandonBrowserRefreshAdmission(ctx context.Context, proposed value.RefreshOwnerRef) error {
	if s == nil || ctx == nil || ctx.Err() != nil || !s.binding.Owns(proposed) {
		return value.ErrDenied
	}
	ctx, cancel := context.WithTimeout(ctx, 5*time.Second)
	defer cancel()
	return refreshError(s.databases.Maintenance.ledger.transaction(ctx, func(q *custodydb.Queries) error {
		if _, e := q.LockBrowserRoot(ctx, id(proposed.RootID)); e != nil {
			return e
		}
		owner, e := q.LockBrowserRefreshOwner(ctx, custodydb.LockBrowserRefreshOwnerParams{RootID: id(proposed.RootID), SessionID: proposed.SessionID[:]})
		if errors.Is(e, pgx.ErrNoRows) {
			return nil
		}
		if e != nil {
			return e
		}
		actual := refreshRef(owner)
		if actual.OwnerID != proposed.OwnerID {
			return nil
		} // Never close a different replayed/current owner.
		if !s.binding.Owns(actual) || actual.Epoch != proposed.Epoch || len(owner.Claim) != 0 {
			return value.ErrDenied
		}
		n, e := q.CurrentTime(ctx)
		if e != nil {
			return e
		}
		if ctx.Err() != nil {
			return value.ErrDenied
		}
		return s.closeStoredBrowserRefreshOwner(ctx, q, owner, n.Time)
	}))
}
func (s *BrowserRefreshOwners) CloseBrowserRefreshOwner(ctx context.Context, ref value.RefreshOwnerRef) error {
	return refreshError(s.maintenance(ctx, ref, false, func(q *custodydb.Queries, owner custodydb.BrowserRefreshOwner, _ custodydb.BrowserSession, n, _ time.Time) error {
		return s.closeStoredBrowserRefreshOwner(ctx, q, owner, n)
	}))
}
func (s *BrowserRefreshOwners) closeStoredBrowserRefreshOwner(ctx context.Context, q *custodydb.Queries, owner custodydb.BrowserRefreshOwner, n time.Time) error {

	if len(owner.Claim) > 0 && !owner.ClaimJoinedAt.Valid {
		return value.ErrDenied
	}
	if !owner.JoinedAt.Valid {
		var e error
		owner, e = q.JoinBrowserRefreshOwner(ctx, custodydb.JoinBrowserRefreshOwnerParams{RootID: owner.RootID, SessionID: owner.SessionID, OwnerID: owner.OwnerID, Generation: owner.Generation, JoinedAt: ts(n)})
		if e != nil {
			return e
		}
	}
	consumers, e := q.ReadBrowserRefreshConsumers(ctx, custodydb.ReadBrowserRefreshConsumersParams{OwnerID: owner.OwnerID, OwnerGeneration: pgtype.Int8{Int64: owner.Generation, Valid: true}})
	if e != nil || len(consumers) > value.MaxRefreshConsumers {
		return eOrDenied(e)
	}
	for _, c := range consumers {
		if _, e = q.DetachBrowserRefreshConsumer(ctx, custodydb.DetachBrowserRefreshConsumerParams{DownloadID: c.DownloadID, OwnerID: owner.OwnerID, OwnerGeneration: pgtype.Int8{Int64: owner.Generation, Valid: true}}); e != nil {
			return e
		}
	}
	_, e = q.DeleteJoinedBrowserRefreshOwner(ctx, custodydb.DeleteJoinedBrowserRefreshOwnerParams{RootID: owner.RootID, SessionID: owner.SessionID, OwnerID: owner.OwnerID, Generation: owner.Generation})
	return e
}

// BrowserRefreshPause resolves the owner from the original server-held session
// key. A content GET cannot nominate/create an owner, refresh, extend a claim,
// renew a response lease or turn a generic busy error into scheduling authority.
func (s *BrowserRefreshOwners) BrowserRefreshPause(ctx context.Context, key [32]byte, principal value.Principal, response value.RefreshResponseBinding) (value.RefreshPause, error) {
	var out value.RefreshPause
	if ctx == nil || ctx.Err() != nil || key == ([32]byte{}) || !principal.Valid() || !response.Valid() {
		return out, value.ErrDenied
	}
	started := s.now()
	until := started.Add(5 * time.Second)
	ctx, cancel := context.WithDeadline(ctx, until)
	defer cancel()
	e := s.databases.Reader.ledger.transaction(ctx, func(q *custodydb.Queries) error {
		trust, e := q.ShareFleetTrust(ctx, id(s.binding.InstallationID))
		if e != nil {
			return e
		}
		root, e := q.LockBrowserRoot(ctx, id(s.sessions.root))
		if e != nil {
			return e
		}
		// The exclusive root lock excludes session and owner mutations for this read.
		// Reusing ReadBrowserSession would require unnecessary UPDATE privilege
		// for its additional FOR SHARE lock on the session itself.
		browser, e := q.ReadBrowserRefreshPauseSession(ctx, custodydb.ReadBrowserRefreshPauseSessionParams{RootID: id(s.sessions.root), Epoch: id(s.sessions.epoch), ID: key[:]})
		if e != nil {
			return e
		}
		owner, e := q.ReadBrowserRefreshOwner(ctx, custodydb.ReadBrowserRefreshOwnerParams{RootID: id(s.sessions.root), SessionID: key[:]})
		if e != nil {
			return e
		}
		ref := refreshRef(owner)
		if uid(root.Epoch) != s.sessions.epoch || ref.Epoch != s.sessions.epoch || ref.InstallationID != s.binding.InstallationID || browserRecord(browser).Principal != principal || !refreshOwnerSession(owner, browser, s.now()) || owner.JoinedAt.Valid || len(owner.Claim) != 32 || owner.ClaimDownloadID != id(response.DownloadID) || owner.ClaimResponseID != id(response.ResponseID) || owner.ClaimResponseOwner != id(response.ResponseOwnerID) || owner.ClaimResponseGeneration.Int64 != response.ResponseGeneration {
			return value.ErrDenied
		}
		settled := owner.ClaimJoinedAt.Valid
		if settled {
			if browser.Version <= owner.ClaimVersion.Int64 || len(browser.RefreshClaim) != 0 {
				return value.ErrDenied
			}
		} else if owner.ClaimVersion.Int64 != browser.Version || !bytes.Equal(owner.Claim, browser.RefreshClaim) || !owner.ClaimUntil.Time.Equal(browser.ClaimUntil.Time) {
			return value.ErrDenied
		}
		consumer, e := q.ReadBrowserRefreshConsumer(ctx, id(response.DownloadID))
		if e != nil {
			return e
		}
		if consumer.OwnerID != owner.OwnerID || consumer.OwnerGeneration.Int64 != owner.Generation || consumer.DownloadRootID != id(s.downloads.root) || !bytes.Equal(consumer.ActorKey, owner.ActorKey) {
			return value.ErrDenied
		}
		exact, e := q.ReadBrowserRefreshPauseResponse(ctx, custodydb.ReadBrowserRefreshPauseResponseParams{ID: id(response.ResponseID), SessionID: id(response.DownloadID), OwnerID: id(response.ResponseOwnerID), Generation: response.ResponseGeneration})
		if e != nil {
			return e
		}
		row, e := q.ReadDownloadSession(ctx, custodydb.ReadDownloadSessionParams{ID: id(response.DownloadID), RootID: id(s.downloads.root)})
		if e != nil {
			return e
		}
		d, e := decodeDownload(row)
		if e != nil || d.Scope.Principal != principal || d.Generation != exact.Generation || d.State.Terminal() {
			return value.ErrDenied
		}
		published, e := q.ReadPublishedContext(ctx, custodydb.ReadPublishedContextParams{ArtifactID: id(d.Scope.ArtifactID), RootID: id(s.downloads.root)})
		if e != nil {
			return e
		}
		n, e := refreshCurrent(ctx, q, root, browser, until)
		if e != nil {
			return e
		}
		recovery, e := recoveryState(trust, started, n)
		if e != nil {
			return e
		}
		claimUntil := value.Earlier(owner.ClaimUntil.Time, value.Earlier(owner.LeaseUntil.Time, value.Earlier(browser.ExpiresAt.Time, value.Earlier(consumer.HardUntil.Time, value.Earlier(exact.UntilAt.Time, value.Earlier(d.Deadline, published.EffectiveExpiresAt.Time))))))
		out = value.RefreshPause{Settled: settled, Owner: ref, DownloadID: response.DownloadID, ResponseID: response.ResponseID, ResponseOwnerID: response.ResponseOwnerID, ResponseGeneration: response.ResponseGeneration, CredentialVersion: owner.ClaimVersion.Int64, ClaimUntil: claimUntil, ObservedAt: started, ValidUntil: value.Earlier(until, value.Earlier(root.FreshUntil.Time, value.Earlier(recovery.FreshUntil, claimUntil)))}
		if !out.Valid(n) {
			return value.ErrDenied
		}
		return nil
	})
	if e != nil || ctx.Err() != nil || !out.Valid(s.now()) {
		return value.RefreshPause{}, refreshError(eOrDenied(e))
	}
	return out, nil
}
