from pathlib import Path
import hashlib,json,os,signal,subprocess,time
stage=Path(__file__).resolve().parent
repo=Path('/home/phil/Projekte/asservat')
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest() if p.is_file() else None
maps={name:json.loads((stage/(name+'.json')).read_text()) for name in ['payload','bases','dependencies']}
def check():
    actual=set(subprocess.run(['rg','--files','-g','*.go','-g','go.mod','-g','go.sum'],cwd=repo,capture_output=True,text=True,check=True).stdout.splitlines())
    expected={k for k in maps['dependencies'] if k.endswith('.go') or Path(k).name in {'go.mod','go.sum'}}
    payloads={str(p.relative_to(stage/'files')) for p in (stage/'files').rglob('*') if p.is_file()}
    return {'live':[k for k,v in maps['dependencies'].items() if sha(repo/k)!=v], 'bases':[k for k,v in maps['bases'].items() if sha(repo/k)!=v], 'payload':[k for k,v in maps['payload'].items() if sha(stage/'files'/k)!=v], 'fileSet':sorted(actual^expected), 'payloadSet':sorted(payloads^set(maps['payload'])), 'mirror':[k for k,v in json.loads((stage/'mirror-manifest.json').read_text()).items() if sha(stage/'architecture-mirror'/k)!=v], 'mirrorSet':sorted({str(p.relative_to(stage/'architecture-mirror')) for p in (stage/'architecture-mirror').rglob('*') if p.is_file()} ^ set(json.loads((stage/'mirror-manifest.json').read_text()))), 'reusedEvidence':[k for k,v in json.loads((stage/'reused-evidence.json').read_text()).items() if sha(Path(k))!=v], 'freeze':[k for k,v in json.loads((stage/'freeze.json').read_text()).items() if sha(stage/k)!=v]}
evidence=stage/'evidence'
evidence.mkdir(exist_ok=False)
env=os.environ.copy();env.update(GOMAXPROCS='2',GOFLAGS='-p=2',TMPDIR=str(stage/'tmp'))
# Only the parent-owned external harness may enable real service tests.
for name in list(env):
    if name.startswith('ASSERVAT_') and ('FIXTURE' in name or 'NATIVE' in name):
        del env[name]
reports=[]
def phase(name,command,expected=0,marker=None):
    before=check()
    if any(before.values()):
        (evidence/(name+'-preflight.json')).write_text(json.dumps(before,indent=2)+'\n')
        raise SystemExit('immutable preflight failed')
    start=time.monotonic()
    with (evidence/(name+'.log')).open('x') as log:
        p=subprocess.Popen(command,cwd=repo,env=env,stdout=log,stderr=subprocess.STDOUT,start_new_session=True)
        print(json.dumps({'phase':name,'pid':p.pid,'command':command}),flush=True)
        try: code=p.wait(timeout=300)
        except subprocess.TimeoutExpired:
            os.killpg(p.pid,signal.SIGTERM)
            try:p.wait(timeout=5)
            except subprocess.TimeoutExpired:os.killpg(p.pid,signal.SIGKILL);p.wait()
            code=124
    after=check()
    occurrences=(evidence/(name+'.log')).read_text().count(marker) if marker else None
    result={'phase':name,'command':command,'cwd':str(repo),'exit':code,'expectedExit':expected,'expectedFailureOccurrences':occurrences,'seconds':round(time.monotonic()-start,3),'before':before,'after':after}
    (evidence/(name+'.json')).write_text(json.dumps(result,indent=2)+'\n');reports.append(result)
    print(json.dumps(result),flush=True)
    if code!=expected or any(after.values()) or marker and occurrences!=2:raise SystemExit(1)
phase('go-architecture',['python3',str(stage/'architecture-mirror/scripts/check-go-architecture.py')])
print(json.dumps({'terminal':0,'finalHashes':check(),'reusedGates':['compile','race','vet']}),flush=True)
