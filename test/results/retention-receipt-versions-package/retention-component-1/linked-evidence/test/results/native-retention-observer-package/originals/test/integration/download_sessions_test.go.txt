// Identity and current regional/recovery qualification are explicitly synthetic.
// The gate, catalog, permission, retained-policy and SQL effect checks are real.
type downloadTestIdentity struct{ value identity.Identity }

func (i downloadTestIdentity) Resolve(_ context.Context, credential string) (identity.Identity, error) {
	if credential != "synthetic-reference" {
		return identity.Identity{}, identity.ErrDenied
	}
	return i.value, nil
}

// This fixed-lifetime synthetic provider supports the longer native fixture.
// It returns a fresh bounded observation, not a two-minute Identity value.
// Product identity/source/session behavior is exercised in its separate suite.
type nativeDownloadTestIdentity struct {
	value   identity.Identity
	expires time.Time
}

func (i nativeDownloadTestIdentity) Resolve(ctx context.Context, credential string) (identity.Identity, error) {
	now := time.Now().UTC()
	if ctx.Err() != nil || credential != "synthetic-reference" || !now.Before(i.expires) {
		return identity.Identity{}, identity.ErrDenied
	}
	observed := i.value
	observed.Groups, observed.Scopes = slices.Clone(i.value.Groups), slices.Clone(i.value.Scopes)
	observed.VerifiedAt, observed.ValidUntil = now, identity.Earlier(now.Add(5*time.Second), i.expires)
	return observed, nil
}

type downloadTestEnvironment struct {
	root, installation, epoch string
	registration              artifact.Registration
	recovery                  *postgres.RecoveryReader
	processing                *postgres.ProcessingReads
	path                      *artifact.DownloadPath
}

func (e downloadTestEnvironment) DownloadEnvironment(ctx context.Context, record authorizationapp.ContextDownloadEvidence, p identity.Principal) (artifact.DownloadEnvironment, error) {
	r := record.Retained.Registration
	if r != e.registration || e.processing == nil || e.path == nil {
		return artifact.DownloadEnvironment{}, artifact.ErrDownloadUnavailable
	}
	// Only the independently accepted installation input is synthetic here.
	// The policy evaluator, recovery reads and durable processing checks use
	// the actual production composition, including its final rereads.
	policy := processingTestPolicy{privacyvalue.DownloadPolicy{RootID: e.root, InstallationID: e.installation, RecoveryEpoch: e.epoch, Revision: identity.Revision{UID: e.path.ID, Generation: 1, Digest: artifact.Digest{87}}, ProtectionDigest: artifact.Digest{88}, PermittedRegions: []string{"synthetic-region"}, Paths: []privacyvalue.DownloadPath{*e.path}, Clients: []privacyvalue.DownloadClient{{Issuer: p.Issuer, ID: p.ClientID, Kind: p.Kind, DataClasses: []string{"metadata"}, PrivacyClasses: []string{"potentialPersonalData"}, EvidenceTypes: []string{"context"}, Purposes: []string{record.PurposeRef}, PathIDs: []string{e.path.ID}}}, Operators: []privacyvalue.ProcessingOperator{{Issuer: p.Issuer, ClientID: p.ClientID, Kind: "Group", ID: "synthetic-privacy"}}}}
	return (authorizationapp.CurrentDownloadEnvironment{RootID: e.root, InstallationID: e.installation, Policies: policy, Recovery: e.recovery, Processing: e.processing, Now: time.Now}).DownloadEnvironment(ctx, record, p)
}

			}
			select {
			case <-completed:
			case <-time.After(5 * time.Second):
				t.Fatal("server accounting did not finish")
			}
			return response.StatusCode, b
		}
		if status, b := get("bytes=5-", "\"wrong-version\""); status != 412 || len(b) != 0 {
			t.Fatal("mismatch disclosed file", status)
		}
		status, partial := get("bytes=5-", metadata.ETag)
		if status != 206 || int64(len(partial)) != metadata.Size-5 {
			t.Fatal("wrong partial", status)
		}
		status, body := get("", metadata.ETag)
		if status != 200 || int64(len(body)) != metadata.Size || sha256.Sum256(body) != metadata.SHA256 || !json.Valid(body) {
			t.Fatal("native JSON/hash mismatch", status)
		}
		if string(partial) != string(body[5:]) {
			t.Fatal("resume did not preserve native bytes")
		}
		// Synthetic local-file workflow, outside managed evidence storage.
		local := filepath.Join(t.TempDir(), metadata.Filename)
		if err = os.WriteFile(local, body, 0600); err != nil {
			t.Fatal(err)
		}
		saved, err := os.ReadFile(local)
		if err != nil || sha256.Sum256(saved) != metadata.SHA256 {
			t.Fatal("local checksum", err)
		}
		clear(saved)
		clear(body)
		clear(partial)
		final, err := f.runtime.LookupDownload(ctx, s.ID, s.Scope.Principal)
		if err != nil || final.State != artifact.DownloadCompleted || final.ChargedRead != 3*evidence.ContextDownloadReadWork || final.EmittedBytes != 2*metadata.Size-5 {
			t.Fatal("final accounting", final.State, err)
		}
	})

// Owned native destination for SQL/TLS/S3 qualification; never a product fallback.
type downloadBytesDestination struct{ w *bytes.Buffer }

func (d downloadBytesDestination) Send(ctx context.Context, _ evidence.DownloadMetadata, _ evidence.DownloadSelection, b []byte) (int64, error) {
	if err := ctx.Err(); err != nil {
		return 0, err
	}
	n, e := d.w.Write(b)
	return int64(n), e
}
