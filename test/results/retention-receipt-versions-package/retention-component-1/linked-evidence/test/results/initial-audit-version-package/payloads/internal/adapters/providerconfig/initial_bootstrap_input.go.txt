// CompileInitialBootstrapInput is local normalization, not authentication or
// acceptance. Actual administrator/Pod/token/guard/API checks remain mandatory.
// The optional token audience is a hint only; no other claim is returned.
func CompileInitialBootstrapInput(v contracts.InitialBootstrapInput, kubernetesCA, token []byte, now time.Time) (contracts.InitialBootstrap, error) {
	var zero contracts.InitialBootstrap
	if v.Format != contracts.InitialBootstrapInputFormat || v.RegistryAuthentication != "" && v.RegistryAuthentication != contracts.InitialRegistryAuthentication || now.IsZero() || len(validation.IsDNS1123Label(v.RunnerAccount)) != 0 || len(validation.IsDNS1123Label(v.BootstrapNamespace)) != 0 {
		return zero, cfg.ErrInvalid
	}
	kubePEM := []byte(v.Kubernetes.RootsPEM)
	if len(kubePEM) == 0 {
		kubePEM = kubernetesCA
	}
	k, err := initialBootstrapPEM(kubePEM, now)
	if err != nil {
		return zero, err
	}
	p, err := initialBootstrapPEM([]byte(v.Postgres.RootsPEM), now)
	if err != nil {
		return zero, err
	}
	a, err := initialBootstrapPEM([]byte(v.AuditStore.RootsPEM), now)
	if err != nil {
		return zero, err
	}
	b, err := initialBootstrapPEM([]byte(v.KeyService.RootsPEM), now)
	if err != nil {
		return zero, err
	}
	audience := v.ProbeAudience
	if audience == "" {
		audience, err = initialBootstrapAudienceHint(token, now)
		if err != nil {
			return zero, err
		}
	}
	if !identity.Text(audience, 256) {
		return zero, cfg.ErrInvalid
	}
	d := contracts.InitialBootstrap{Format: contracts.InitialBootstrapVersionedAuditFormat, Name: v.Name, BootstrapNamespace: v.BootstrapNamespace, TargetNamespace: v.TargetNamespace, Image: v.Image, RegistryAuthentication: v.RegistryAuthentication, Administrators: []contracts.OwnerDeploymentActor{{Kind: "User", Name: "system:serviceaccount:" + v.BootstrapNamespace + ":" + v.RunnerAccount}}, ProbeAudience: audience, Kubernetes: contracts.OwnerProviderEndpoint{Endpoint: v.Kubernetes.Endpoint, Roots: k}, Postgres: contracts.OwnerPostgresEndpoint{Host: v.Postgres.Host, Port: v.Postgres.Port, Database: v.Postgres.Database, MaxConnections: v.Postgres.MaxConnections, Roots: p}, AuditStore: contracts.OwnerAuditStore{Endpoint: v.AuditStore.Endpoint, Region: v.AuditStore.Region, Bucket: v.AuditStore.Bucket, Roots: a}, KeyService: contracts.ConfigurationKeyService{Endpoint: v.KeyService.Endpoint, AuditMount: v.KeyService.AuditMount, AuditKey: v.KeyService.AuditKey, Roots: b}, Fleet: v.Fleet, Providers: v.Providers}
	if Validate(d.Kubernetes, d.Postgres, d.AuditStore, d.KeyService, now) != nil || !InitialProviderNames(d.Providers.IssuerPolicy, d.Providers.WrapRole, d.Providers.WrapPolicy, d.Providers.UnwrapRole, d.Providers.UnwrapPolicy) {
		return zero, cfg.ErrInvalid
	}
	encoded, err := json.Marshal(d)
	if err != nil || len(encoded) > contracts.InitialBootstrapMaxBytes {
		return zero, cfg.ErrInvalid
	}
	return d, nil
}

func initialBootstrapPEM(raw []byte, now time.Time) ([][]byte, error) {
	if len(raw) == 0 || len(raw) > 32<<10 {
		return nil, cfg.ErrInvalid
	}
	var roots [][]byte
	for rest := bytes.TrimSpace(raw); len(rest) > 0; {
		// pem.Decode skips preceding junk; that would silently ignore private
		// keys or contradictory input. Only exact certificate blocks are legal.
		if !bytes.HasPrefix(rest, []byte("-----BEGIN CERTIFICATE-----")) {
			return nil, cfg.ErrInvalid
		}
		block, next := pem.Decode(rest)
		if block == nil || block.Type != "CERTIFICATE" || len(block.Headers) != 0 || len(roots) == 4 {
			return nil, cfg.ErrInvalid
		}
		roots = append(roots, append([]byte(nil), block.Bytes...))
		rest = bytes.TrimSpace(next)
	}
	if _, _, err := mtls.OwnerTrustRoots(roots, now); err != nil {
		return nil, cfg.ErrInvalid
	}
	slices.SortFunc(roots, bytes.Compare)
	return roots, nil
}

// No signature is checked here because this cannot confer authority. The
// resulting audience is subsequently used in actual independent API checks.
func initialBootstrapAudienceHint(token []byte, now time.Time) (string, error) {
	if len(token) == 0 || len(token) > 16<<10 {
		return "", cfg.ErrInvalid
	}
	parts := strings.Split(string(token), ".")
	if len(parts) != 3 || parts[0] == "" || parts[2] == "" {
		return "", cfg.ErrInvalid
	}
	payload, err := base64.RawURLEncoding.DecodeString(parts[1])
	defer clear(payload)
	if err != nil || base64.RawURLEncoding.EncodeToString(payload) != parts[1] || !jsontext.Value(payload).IsValid() {
		return "", cfg.ErrInvalid
	}
	var claims map[string]json.RawMessage
	if json.Unmarshal(payload, &claims) != nil {
		return "", cfg.ErrInvalid
	}
	var exp int64
	if json.Unmarshal(claims["exp"], &exp) != nil || exp <= now.Unix() {
		return "", cfg.ErrInvalid
	}
	if raw, exists := claims["nbf"]; exists {
		var nbf int64
		if json.Unmarshal(raw, &nbf) != nil || nbf > now.Unix() {
			return "", cfg.ErrInvalid
		}
	}
	var audience string
	if json.Unmarshal(claims["aud"], &audience) != nil {
		var all []string
		if json.Unmarshal(claims["aud"], &all) != nil || len(all) != 1 {
			return "", cfg.ErrInvalid
		}
		audience = all[0]
	}
	if !identity.Text(audience, 256) {
		return "", cfg.ErrInvalid
	}
	return audience, nil
}
