func TestWorkerNativeJoinedDoesNotSpendNextCommandBudgetOnIdleWait(t *testing.T) {
	now := time.Now().UTC()
	id := "11111111-1111-4111-8111-111111111111"
	target := value.WorkerCredentialTarget{InstallationID: id, SessionID: id, WorkerID: id, PodUID: id, CreatedAt: now, RunUntil: now.Add(time.Minute)}
	// The next independently admitted command has a shorter original interval.
	// The worker cannot know or renew it; this models its receive-side cutoff.
	nextCommandUntil := now.Add(500 * time.Millisecond)
	tokens := &nativeTokenFixture{}
	stream := &nativeStreamFixture{ctx: t.Context()}
	opens, waits := 0, 0
	firstNonce := ""
	loop := workerNativeLoop{target: target, token: tokens, now: func() time.Time { return now }, nonce: workerNativeNonce,
		wait: func(_ context.Context, d time.Duration) error { waits++; now = now.Add(d); return nil },
		open: func(ctx context.Context, h contracts.NativeHello) (contracts.NativeStream, error) {
			opens++
			if deadline, ok := ctx.Deadline(); !ok || !deadline.Equal(target.RunUntil) {
				t.Fatal("original worker end changed")
			}
			if opens == 1 {
				firstNonce = h.Nonce
				return stream, nil
			}
			if !now.Before(nextCommandUntil) {
				t.Fatal("positive Join consumed the next original command interval in idle backoff")
			}
			if tokens.reads != 2 || h.Nonce == firstNonce || !contracts.ValidWorkerNonce(h.Nonce) || stream.closed != 1 {
				t.Fatal("next Hello did not use fresh token/nonce after joined prior close")
			}
			for _, b := range tokens.values[0] {
				if b != 0 {
					t.Fatal("prior projected token survived into next Hello")
				}
			}
			return nil, io.ErrUnexpectedEOF // Unknown second setup must stop, never retry.
		},
		run: func(context.Context, contracts.NativeStream, contracts.NativeHello) (contracts.NativeConversationResult, error) {
			return contracts.NativeConversationResult{State: contracts.NativeJoined}, nil
		},
	}
	if err := loop.Run(t.Context()); !errors.Is(err, artifact.ErrDownloadUnavailable) || opens != 2 || waits != 0 {
		t.Fatalf("unexpected selection after joined command: opens=%d waits=%d err=%v", opens, waits, err)
	}
}

func TestWorkerNativeJoinedKeepsOriginalStopBounds(t *testing.T) {
	for _, mode := range []string{"parent-cancelled", "worker-expired", "shorter-parent"} {
		t.Run(mode, func(t *testing.T) {
			now := time.Now().UTC()
			id := "11111111-1111-4111-8111-111111111111"
			target := value.WorkerCredentialTarget{InstallationID: id, SessionID: id, WorkerID: id, PodUID: id, CreatedAt: now, RunUntil: now.Add(time.Minute)}
			parentUntil := now.Add(10 * time.Second)
			ctx, cancel := context.WithDeadline(t.Context(), parentUntil)
			defer cancel()
			tokens := &nativeTokenFixture{}
			stream := &nativeStreamFixture{ctx: ctx}
			opens, waits := 0, 0
			loop := workerNativeLoop{target: target, token: tokens, now: func() time.Time { return now }, nonce: workerNativeNonce,
				wait: func(context.Context, time.Duration) error { waits++; return nil },
				open: func(ctx context.Context, _ contracts.NativeHello) (contracts.NativeStream, error) {
					opens++
					if deadline, ok := ctx.Deadline(); !ok || !deadline.Equal(parentUntil) {
						t.Fatal("shorter original parent interval changed")
					}
					if opens > 1 {
						return nil, io.ErrUnexpectedEOF
					}
					return stream, nil
				},
				run: func(context.Context, contracts.NativeStream, contracts.NativeHello) (contracts.NativeConversationResult, error) {
					switch mode {
					case "parent-cancelled":
						cancel()
					case "worker-expired":
						now = target.RunUntil
					}
					return contracts.NativeConversationResult{State: contracts.NativeJoined}, nil
				},
			}
			err := loop.Run(ctx)
			wantOpens := 1
			wantError := artifact.ErrDownloadExpired
			if mode == "shorter-parent" {
				wantOpens = 2
				wantError = artifact.ErrDownloadUnavailable
			}
			if !errors.Is(err, wantError) || opens != wantOpens || tokens.reads != wantOpens || waits != 0 || stream.closed != 1 {
				t.Fatalf("original stop bound changed: opens=%d tokens=%d waits=%d closes=%d err=%v", opens, tokens.reads, waits, stream.closed, err)
			}
		})
	}
}

func TestWorkerNativeConversationCeilingAppliesToJoinedAndIdle(t *testing.T) {
	for _, idle := range []bool{false, true} {
		name := "joined"
		if idle {
			name = "idle"
		}
		t.Run(name, func(t *testing.T) {
			now := time.Now().UTC()
			id := "11111111-1111-4111-8111-111111111111"
			target := value.WorkerCredentialTarget{InstallationID: id, SessionID: id, WorkerID: id, PodUID: id, CreatedAt: now, RunUntil: now.Add(time.Minute)}
			tokens := &nativeTokenFixture{}
			opens, runs, waits := 0, 0, 0
			var previous *nativeStreamFixture
			loop := workerNativeLoop{target: target, token: tokens, now: func() time.Time { return now }, nonce: workerNativeNonce,
				wait: func(_ context.Context, d time.Duration) error {
					waits++
					if d != time.Second {
						t.Fatal("idle backoff changed")
					}
					return nil
				},
				open: func(ctx context.Context, h contracts.NativeHello) (contracts.NativeStream, error) {
					opens++
					if previous != nil && previous.closed != 1 {
						t.Fatal("next connection preceded prior close")
					}
					if idle {
						return nil, &contracts.NativeNoCommand{Reference: h.Reference, Nonce: h.Nonce, Until: now.Add(time.Second)}
					}
					previous = &nativeStreamFixture{ctx: ctx}
					return previous, nil
				},
				run: func(context.Context, contracts.NativeStream, contracts.NativeHello) (contracts.NativeConversationResult, error) {
					runs++
					return contracts.NativeConversationResult{State: contracts.NativeJoined}, nil
				},
			}
			if err := loop.Run(t.Context()); !errors.Is(err, artifact.ErrDownloadUnavailable) {
				t.Fatalf("ceiling result: %v", err)
			}
			wantRuns, wantWaits := contracts.NativeMaximumConversations, 0
			if idle {
				wantRuns, wantWaits = 0, contracts.NativeMaximumConversations
			}
			if opens != contracts.NativeMaximumConversations || tokens.reads != opens || runs != wantRuns || waits != wantWaits {
				t.Fatalf("conversation slots changed: opens=%d tokens=%d runs=%d waits=%d", opens, tokens.reads, runs, waits)
			}
		})
	}
}
