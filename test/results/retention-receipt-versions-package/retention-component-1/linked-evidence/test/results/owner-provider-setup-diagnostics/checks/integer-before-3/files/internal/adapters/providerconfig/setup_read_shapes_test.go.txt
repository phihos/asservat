// Actual public responses from the owned pinned OpenBao 2.7.1 read-shape probe.
// No reviewer JWT, access token or private key is retained. The probe's fake
// reviewer does not qualify authentication; separate Kubernetes tests do that.
func TestOwnerProviderSetupDecodesQualifiedProviderShapes(t *testing.T) {
	var fixture struct {
		Recipe       OwnerSetupRecipe
		AuthAccessor string
		Observations []struct {
			Path       string
			PublicData map[string]json.RawMessage
		}
	}
	if e := json.Unmarshal([]byte(qualifiedSetupResponses), &fixture); e != nil {
		t.Fatal(e)
	}
	var config map[string]json.RawMessage
	if json.Unmarshal(fixture.Recipe.AuthConfiguration, &config) != nil {
		t.Fatal("expected auth config")
	}
	host, _ := setupValue[string](config, "kubernetes_host")
	ca, _ := setupValue[string](config, "kubernetes_ca_cert")
	roots, e := setupPEMCertificates(ca)
	if e != nil {
		t.Fatal(e)
	}
	checks := map[string]func(map[string]json.RawMessage) error{
		"sys/mounts/auth/owner-auth": func(d map[string]json.RawMessage) error { return verifyAuthMount(d, fixture.AuthAccessor) },
		"auth/owner-auth/config": func(d map[string]json.RawMessage) error {
			return verifyKubernetesAuthConfig(d, contracts.OwnerProviderEndpoint{Endpoint: host, Roots: roots})
		},
		"sys/mounts/transit": verifyTransitMount,
		"transit/keys/audit": func(d map[string]json.RawMessage) error { return verifyTransitKey(d, "audit") },
		"sys/mounts/pki": func(d map[string]json.RawMessage) error {
			if !setupIs(d, "type", "pki") {
				return ErrSetupUnavailable
			}
			return nil
		},
		"pki/cert/ca": func(d map[string]json.RawMessage) error {
			raw, _ := setupValue[string](d, "certificate")
			r, e := setupPEMCertificates(raw)
			if e != nil {
				return e
			}
			cert, e := x509.ParseCertificate(r[0])
			if e != nil {
				return e
			}
			_, e = verifySetupIssuer(d, r, cert.NotBefore.Add(time.Minute))
			return e
		},
	}
	for _, role := range fixture.Recipe.Authentication.Roles {
		checks["auth/owner-auth/role/"+role.Name] = func(d map[string]json.RawMessage) error { return verifyKubernetesRole(d, role.Role) }
		checks["sys/policies/acl/"+role.Name] = func(d map[string]json.RawMessage) error { return verifyACL(d, role.Name, role.Policy) }
	}
	for _, role := range fixture.Recipe.PKIRoles {
		checks["pki/roles/"+role.Name] = func(d map[string]json.RawMessage) error { return verifySetupPKIRole(d, role.Role) }
	}
	if len(checks) != 16 || len(fixture.Observations) != 16 {
		t.Fatal("incomplete captured fixture")
	}
	for _, row := range fixture.Observations {
		t.Run(row.Path, func(t *testing.T) {
			check, ok := checks[row.Path]
			if !ok {
				t.Fatal("unqualified response")
			}
			if e := check(row.PublicData); e != nil {
				t.Fatal(e)
			}
		})
	}
}
